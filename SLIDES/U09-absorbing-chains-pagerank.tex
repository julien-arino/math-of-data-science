\begin{frame}{From Unit 08 to Unit 09}
In Unit 08, we studied irreducible and primitive non-negative matrices and saw that a regular Markov chain converges to a unique positive stationary distribution $\bw = P\bw$
\vfill
Two natural questions arise when a directed network is \emph{not} regular:
\begin{enumerate}
\item What happens when a Markov chain has \defword{absorbing states} (states that can be entered but never left), making $P$ reducible?
\item How do we rank the importance of vertices in large directed networks (citations, social networks or the World Wide Web) using eigenvectors and linear systems?
\end{enumerate}
\vfill
Both questions are answered by the same two linear algebra tools: \textbf{Perron-Frobenius theory} and the \textbf{Neumann series} $(\II - M)^{-1} = \sum_{k=0}^\infty M^k$
\end{frame}

\begin{frame}{Absorbing states and absorbing Markov chains}
\begin{importantdefinition}[Absorbing state]
A state $S_i$ in a Markov chain with column-stochastic transition matrix $P = [p_{ij}]$ is \defword{absorbing} if once the chain enters $S_i$, it never leaves:
\[
p_{ii} = 1 \quad \text{and} \quad p_{ji} = 0 \text{ for all } j \neq i
\]
Equivalently, column $i$ of $P$ is the standard basis vector $\be_i$, and in $\G(P)$ vertex $S_i$ has a self-loop of weight $1$ and no outgoing arcs to any other vertex
\end{importantdefinition}
\vfill
\begin{importantdefinition}[Absorbing Markov chain]
A Markov chain is \defword{absorbing} if:
\begin{enumerate}
\item It has at least one absorbing state
\item From every non-absorbing state, it is possible to reach at least one absorbing state in a finite number of steps (there is a directed walk in $\G(P)$ to an absorbing state)
\end{enumerate}
In an absorbing Markov chain, a state that is not absorbing is called \defword{transient}
\end{importantdefinition}
\end{frame}

\SSsubsection{Canonical block form and reducibility}

\begin{frame}{Absorbing chains are reducible}
Suppose a Markov chain on $n$ states has $r \ge 1$ absorbing states and $m = n - r \ge 1$ transient states
\vfill
\begin{itemize}
\item Because an absorbing state $S_i$ has no outgoing arcs to any other vertex ($p_{ji} = 0$ for all $j \neq i$), there is no directed path from $S_i$ to any transient state
\item Therefore, the transition digraph $\G(P)$ is \textbf{not strongly connected}
\item By the characterisation from Unit 08, the transition matrix $P$ is \defword{reducible}
\end{itemize}
\vfill
Relabelling the states of the chain corresponds to applying a permutation matrix $\Pi \in \M_n$:
\[
\tilde{P} = \Pi^T P \Pi
\]
We always relabel the states so that the $r$ \textbf{absorbing states come first} ($S_1, \ldots, S_r$) and the $m = n - r$ \textbf{transient states come last} ($S_{r+1}, \ldots, S_n$)
\end{frame}

\begin{frame}{Example -- Reordering the 5-state random walk into canonical form}
In Motivating example 1 on $(S_1, S_2, S_3, S_4, S_5)$, states $S_1$ and $S_5$ are absorbing while $S_2, S_3, S_4$ are transient. In the original state ordering, the transition matrix is
\[
P_{\text{orig}} = \begin{array}{c@{\hspace{4pt}}c}
& \begin{array}{ccccc} S_1 & S_2 & S_3 & S_4 & S_5 \end{array} \\
\begin{array}{c} S_1 \\ S_2 \\ S_3 \\ S_4 \\ S_5 \end{array} &
\left(\begin{array}{ccccc}
1 & q & 0 & 0 & 0 \\
0 & 0 & q & 0 & 0 \\
0 & p & 0 & q & 0 \\
0 & 0 & p & 0 & 0 \\
0 & 0 & 0 & p & 1
\end{array}\right)
\end{array}
\]
Reordering the states as $(S_1, S_5, S_2, S_3, S_4)$ via the permutation matrix $\Pi = [\be_1\; \be_5\; \be_2\; \be_3\; \be_4]$ gives the canonical block form $\tilde{P} = \Pi^T P_{\text{orig}} \Pi$:
\[
\tilde{P} = \begin{array}{c@{\hspace{4pt}}c}
& \begin{array}{cc|ccc} S_1 & S_5 & S_2 & S_3 & S_4 \end{array} \\
\begin{array}{c} S_1 \\ S_5 \\ \hline S_2 \\ S_3 \\ S_4 \end{array} &
\left(\begin{array}{cc|ccc}
1 & 0 & q & 0 & 0 \\
0 & 1 & 0 & 0 & p \\ \hline
0 & 0 & 0 & q & 0 \\
0 & 0 & p & 0 & q \\
0 & 0 & 0 & p & 0
\end{array}\right)
\end{array} = \begin{pmatrix} \II_2 & R \\ \b0_{3,2} & Q \end{pmatrix}
\]
\end{frame}

\begin{frame}{Canonical block form of an absorbing chain}
\begin{importantdefinition}[Canonical block form]
For an absorbing Markov chain with $r$ absorbing states (listed first) and $m = n - r$ transient states (listed last), the column-stochastic transition matrix takes the \defword{canonical block form} (or \textbf{standard form})
\[
P = \begin{pmatrix}
\II_r & R \\
\b0_{m,r} & Q
\end{pmatrix}
\]
where:
\begin{itemize}
\item $\II_r \in \M_r$ is the $r \times r$ identity matrix (absorbing states stay put)
\item $\b0_{m,r} \in \M_{m,r}$ is the $m \times r$ zero matrix (absorbing states never move to transient states)
\item $R \in \M_{r,m}$ contains the one-step transition probabilities from transient states to absorbing states
\item $Q \in \M_m$ contains the one-step transition probabilities among transient states
\end{itemize}
\end{importantdefinition}
\end{frame}

\begin{frame}{Visualising the canonical blocks}
In our column-stochastic convention, columns correspond to the \textbf{from} state (at time $t$) and rows correspond to the \textbf{to} state (at time $t+1$):
\vfill
\begin{center}
\renewcommand{\arraystretch}{1.5}
\begin{tabular}{c|cc}
& \textbf{From absorbing} ($r$ cols) & \textbf{From transient} ($m$ cols) \\ \hline
\textbf{To absorbing} ($r$ rows) & $\II_r \in \M_r$ & $R \in \M_{r,m}$ \\
\textbf{To transient} ($m$ rows) & $\b0_{m,r} \in \M_{m,r}$ & $Q \in \M_m$
\end{tabular}
\end{center}
\vfill
Notice that:
\begin{itemize}
\item Each column of $\begin{pmatrix} R \\ Q \end{pmatrix}$ sums to $1$: $\11_r^T R + \11_m^T Q = \11_m^T$
\item $Q$ is non-negative and \textbf{column-substochastic}: $\11_m^T Q \le \11_m^T$, with strict inequality in every column $j$ that has a direct transition to an absorbing state ($\sum_{i=1}^r r_{ij} > 0$)
\end{itemize}
\end{frame}

\begin{frame}{Powers of $P$ in canonical form}
To understand the long-term behaviour $\bp(k) = P^k \bp(0)$, we compute the $k$th power of $P$ in block form
\vfill
\begin{proposition}[Powers of the canonical transition matrix]
\label{prop:powers-canonical-P}
Let $P = \begin{pmatrix} \II_r & R \\ \b0_{m,r} & Q \end{pmatrix}$ be in canonical block form. For every integer $k \ge 1$,
\[
P^k = \begin{pmatrix}
\II_r & R(\II_m + Q + Q^2 + \cdots + Q^{k-1}) \\
\b0_{m,r} & Q^k
\end{pmatrix} = \begin{pmatrix}
\II_r & R \sum_{j=0}^{k-1} Q^j \\
\b0_{m,r} & Q^k
\end{pmatrix}
\]
\end{proposition}
\vfill
The $(i,j)$ entry of $Q^k$ is the probability that the chain is in the $i$th transient state after $k$ steps, given that it started in the $j$th transient state
\end{frame}

\begin{frame}[red]{Proof of Proposition~\ref{prop:powers-canonical-P}}
We proceed by mathematical induction on $k \ge 1$
\vfill
\textbf{Base case ($k=1$):} Since $Q^0 = \II_m$, the formula gives $\begin{pmatrix} \II_r & R\II_m \\ \b0_{m,r} & Q^1 \end{pmatrix} = P$, which holds
\vfill
\textbf{Inductive step:} Suppose the formula holds for $k \ge 1$. Using block matrix multiplication for $P^{k+1} = P^k P$:
\begin{align*}
P^{k+1} &= \begin{pmatrix}
\II_r & R(\II_m + Q + \cdots + Q^{k-1}) \\
\b0_{m,r} & Q^k
\end{pmatrix}
\begin{pmatrix}
\II_r & R \\
\b0_{m,r} & Q
\end{pmatrix} \\
&= \begin{pmatrix}
\II_r \II_r + R(\cdots)\b0_{m,r} & \II_r R + R(\II_m + Q + \cdots + Q^{k-1})Q \\
\b0_{m,r}\II_r + Q^k \b0_{m,r} & \b0_{m,r} R + Q^k Q
\end{pmatrix} \\
&= \begin{pmatrix}
\II_r & R(\II_m + Q + Q^2 + \cdots + Q^k) \\
\b0_{m,r} & Q^{k+1}
\end{pmatrix}
\end{align*}
This completes the induction \hfill $\blacksquare$
\end{frame}

\SSsubsection{The Neumann series and fundamental matrix}

\begin{frame}{Spectral radius of the transient block $Q$}
\begin{theorem}[Contraction of the transient block]
\label{th:rho-Q-less-than-1}
In any absorbing Markov chain with canonical transient block $Q \in \M_m$, the spectral radius satisfies
\[
\rho(Q) < 1 \quad \text{and consequently} \quad \lim_{k \to \infty} Q^k = \b0_m
\]
\end{theorem}
\vfill
\textbf{Why does this hold?}
\begin{itemize}
\item By definition of an absorbing chain, from every transient state $j \in \{1,\ldots,m\}$ there is a directed walk to some absorbing state of length at most $m$
\item Thus, starting from transient state $j$, the probability of being absorbed within $m$ steps is strictly positive ($> 0$), so the probability of remaining in the transient states after $m$ steps is strictly less than $1$
\item This means every column sum of $Q^m$ is strictly less than $1$: $\|Q^m\|_1 = \max_j \sum_{i=1}^m (Q^m)_{ij} < 1$
\item Therefore $\rho(Q)^m = \rho(Q^m) \le \|Q^m\|_1 < 1$, which implies $\rho(Q) < 1$ and $\lim_{k\to\infty} Q^k = \b0_m$
\end{itemize}
\end{frame}

\begin{frame}{Absorption occurs with probability 1}
Because $(Q^k)_{ij}$ is the probability of being in transient state $i$ at step $k$ starting from transient state $j$, Theorem~\ref{th:rho-Q-less-than-1} immediately answers our first question
\vfill
\begin{importanttheorem}[Eventual absorption]
In an absorbing Markov chain, the probability that the process eventually reaches an absorbing state is $1$, regardless of the initial state
\end{importanttheorem}
\vfill
Indeed, starting from transient state $j$, the probability of still being in \emph{some} transient state after $k$ steps is the $j$th column sum $\sum_{i=1}^m (Q^k)_{ij} = (\11_m^T Q^k)_j \to 0$ as $k \to \infty$
\end{frame}

\begin{frame}{The Neumann series for matrices}
Recall the scalar geometric series: for $q \in \IR$ with $|q| < 1$, $(1 - q)^{-1} = \sum_{k=0}^\infty q^k$. The exact same identity holds for square matrices!
\vfill
\begin{importanttheorem}[Neumann series]
\label{th:neumann-series}
Let $Q \in \M_m$ be a matrix with spectral radius $\rho(Q) < 1$ (equivalently, $\lim_{k\to\infty} Q^k = \b0_m$). Then $\II_m - Q$ is invertible and its inverse is given by the convergent series
\[
(\II_m - Q)^{-1} = \sum_{k=0}^\infty Q^k = \II_m + Q + Q^2 + Q^3 + \cdots
\]
\end{importanttheorem}
\vfill
In particular, since $Q \ge 0$ for an absorbing Markov chain, every term $Q^k \ge 0$ is non-negative, so $(\II_m - Q)^{-1} \ge \II_m \ge 0$ has strictly non-negative entries (and diagonal entries $\ge 1$)
\end{frame}

\begin{frame}[red]{Proof of Theorem~\ref{th:neumann-series} (Neumann series)}
Consider the partial sum $S_k = \II_m + Q + Q^2 + \cdots + Q^{k-1}$ for $k \ge 1$. Multiplying by $\II_m - Q$ gives a telescoping sum:
\begin{align*}
(\II_m - Q) S_k &= (\II_m - Q)(\II_m + Q + Q^2 + \cdots + Q^{k-1}) \\
&= (\II_m + Q + Q^2 + \cdots + Q^{k-1}) - (Q + Q^2 + \cdots + Q^{k-1} + Q^k) \\
&= \II_m - Q^k
\end{align*}
\vfill
First, suppose $(\II_m - Q)\bx = \b0$. Then $\bx = Q\bx = Q^2\bx = \cdots = Q^k\bx \to \b0$ as $k \to \infty$ since $\rho(Q) < 1$. Hence $\nullspace(\II_m - Q) = \{\b0\}$, so $\II_m - Q$ is invertible
\vfill
Multiplying $(\II_m - Q)S_k = \II_m - Q^k$ on the left by $(\II_m - Q)^{-1}$ gives
\[
S_k = (\II_m - Q)^{-1}(\II_m - Q^k) = (\II_m - Q)^{-1} - (\II_m - Q)^{-1}Q^k
\]
Taking the limit as $k \to \infty$ and using $\lim_{k\to\infty} Q^k = \b0_m$ yields
\[
\sum_{j=0}^\infty Q^j = \lim_{k\to\infty} S_k = (\II_m - Q)^{-1} \qquad \blacksquare
\]
\end{frame}

\begin{frame}{The fundamental matrix and absorption quantities}
\begin{importantdefinition}[Fundamental matrix]
For an absorbing Markov chain with canonical transient block $Q \in \M_m$, the matrix
\[
N = (\II_m - Q)^{-1} = \II_m + Q + Q^2 + \cdots \in \M_m
\]
is called the \defword{fundamental matrix} of the absorbing Markov chain
\end{importantdefinition}
\vfill
\begin{importanttheorem}[Quantitative analysis of absorbing Markov chains]
\label{th:absorbing-chain-quantities}
Let $P = \begin{pmatrix} \II_r & R \\ \b0_{m,r} & Q \end{pmatrix}$ be in canonical form with fundamental matrix $N = [n_{ij}] = (\II_m - Q)^{-1}$. Then:
\begin{enumerate}
\item \textbf{Expected visits to transient states:} $n_{ij}$ is the expected total number of times the chain is in the $i$th transient state, given that it starts in the $j$th transient state
\item \textbf{Expected time to absorption:} Let $\bt^T = \11_m^T N = (t_1, \ldots, t_m)$. Then $t_j = \sum_{i=1}^m n_{ij}$ (the $j$th column sum of $N$) is the expected number of steps before absorption when starting in the $j$th transient state
\item \textbf{Absorption probabilities:} Let $B = [b_{ij}] = R N \in \M_{r,m}$. Then $b_{ij}$ is the probability of eventually being absorbed in the $i$th absorbing state when starting in the $j$th transient state
\end{enumerate}
\end{importanttheorem}
\end{frame}

\begin{frame}[red]{Proof of Theorem~\ref{th:absorbing-chain-quantities}}
\textbf{1. Expected visits $N = (\II_m - Q)^{-1}$:} Starting in transient state $j$, let $X_{ij}^{(k)} \in \{0,1\}$ be the indicator random variable that equals $1$ if the chain is in transient state $i$ at step $k \ge 0$, and $0$ otherwise. Then $\IE[X_{ij}^{(k)}] = \IP(\text{in transient state } i \text{ at step } k \mid \text{start in } j) = (Q^k)_{ij}$. By linearity of expectation, the expected total number of visits to transient state $i$ is
\[
\IE\left[\sum_{k=0}^\infty X_{ij}^{(k)}\right] = \sum_{k=0}^\infty (Q^k)_{ij} = \left(\sum_{k=0}^\infty Q^k\right)_{ij} = n_{ij}
\]
\textbf{2. Time to absorption $\bt^T = \11_m^T N$:} The total number of steps spent in \emph{all} transient states before absorption starting from $j$ is the sum over all $i=1,\ldots,m$ of visits to transient state $i$:
\[
t_j = \sum_{i=1}^m n_{ij} = (\11_m^T N)_j
\]
\textbf{3. Absorption probabilities $B = R N$:} Taking $k \to \infty$ in Proposition~\ref{prop:powers-canonical-P} gives
\[
\lim_{k\to\infty} P^k = \begin{pmatrix} \II_r & R \sum_{j=0}^\infty Q^j \\ \b0_{m,r} & \lim_{k\to\infty} Q^k \end{pmatrix} = \begin{pmatrix} \II_r & R N \\ \b0_{m,r} & \b0_m \end{pmatrix} = \begin{pmatrix} \II_r & B \\ \b0_{m,r} & \b0_m \end{pmatrix} \qquad \blacksquare
\]
\end{frame}

\SSsection{Spectral centrality in networks}

\SSsubsection{Degree and eigenvector centrality}

\begin{frame}{Why spectral centrality?}
Combinatorial centrality measures on graphs are often based on counting shortest paths (such as closeness and betweenness centrality)
\vfill
\textbf{Computational bottleneck on massive networks:}
\begin{itemize}
\item All-pairs shortest-path algorithms require $O(n m)$ or $O(n^3)$ operations
\item For a social network or web graph with $n \sim 10^8$ vertices, $n^2 \sim 10^{16}$ is completely intractable
\end{itemize}
\vfill
\textbf{The spectral alternative:}
\begin{itemize}
\item Express vertex importance via matrix-vector products $A\bx$, eigenvectors $A\bx = \lambda\bx$ or linear systems $(\II - \alpha A)\bx = \bb$
\item When the adjacency matrix $A$ is \textbf{sparse} ($m = |E| \ll n^2$ non-zero entries), one sparse matrix-vector product $A\bx$ takes only $O(m)$ time!
\end{itemize}
\end{frame}

\begin{frame}{Convention for directed adjacency matrices}
To remain consistent with our column-stochastic Markov chain convention $\bp(t+1) = P\bp(t)$ where column $j$ records transitions \emph{from} $j$ \emph{to} $i$, we use the following convention for a directed graph $\G = (V, E)$ on $n$ vertices:
\vfill
\[
a_{ij} = \begin{cases}
1 & \text{if there is a directed arc } j \to i \text{ (from } j \text{ to } i\text{)} \\
0 & \text{otherwise}
\end{cases}
\]
\vfill
With this convention:
\begin{itemize}
\item \textbf{Row $i$} of $A$ records arcs \emph{entering} vertex $i$, so the \defword{in-degree} of vertex $i$ is $d_i^{\text{in}} = \sum_{j=1}^n a_{ij} = (A\11_n)_i$
\item \textbf{Column $j$} of $A$ records arcs \emph{leaving} vertex $j$, so the \defword{out-degree} of vertex $j$ is $d_j^{\text{out}} = \sum_{i=1}^n a_{ij} = (\11_n^T A)_j$
\item For an undirected graph, $A = A^T$ is symmetric and $d_i^{\text{in}} = d_i^{\text{out}} = d_i$
\end{itemize}
\end{frame}

\begin{frame}{First-order measure -- In-degree centrality}
The simplest measure of prestige in a directed network is \defword{in-degree centrality}:
\[
\bd^{\text{in}} = A \11_n, \qquad d_i^{\text{in}} = \sum_{j=1}^n a_{ij}
\]
Each arc $j \to i$ is treated as one ``vote'' for the importance of vertex $i$
\vfill
\textbf{Fundamental limitation:}
\begin{itemize}
\item In-degree treats all incoming links as equal
\item Receiving a link from a major hub or authoritative page counts the exact same ($+1$) as receiving a link from an isolated spam page
\item We want a \textbf{recursive} notion of prestige: \emph{a vertex is important if it is pointed to by other important vertices}
\end{itemize}
\end{frame}

\begin{frame}{Eigenvector centrality}
Instead of summing $1$ for each incoming neighbour $j \to i$, we sum the \emph{importance scores} $x_j$ of those neighbours and require $x_i$ to be proportional to that sum
\vfill
\begin{importantdefinition}[Eigenvector centrality]
Let $A = [a_{ij}] \in \M_n$ be the adjacency matrix of a graph or digraph (with $a_{ij} = 1$ if $j \to i$). The \defword{eigenvector centrality} (or \textbf{prestige score}) of vertex $i$ is the $i$th entry of the non-negative eigenvector $\bx \ge \b0$ satisfying
\[
\lambda x_i = \sum_{j=1}^n a_{ij} x_j, \quad i=1,\ldots,n \iff A \bx = \lambda \bx
\]
associated with the dominant eigenvalue $\lambda = \rho(A)$
\end{importantdefinition}
\vfill
Notice that $x_i = \frac{1}{\lambda}\sum_{j \in \N^{\text{in}}(i)} x_j$ depends on both the \textbf{quantity} of in-neighbours $|\N^{\text{in}}(i)|$ and the \textbf{quality} $x_j$ of each in-neighbour
\end{frame}

\begin{frame}{Why Perron-Frobenius theory is essential here}
For $\bx$ to be a valid centrality ranking, its entries must be \textbf{non-negative} ($x_i \ge 0$) and uniquely determined (up to normalisation $\11_n^T \bx = 1$)
\vfill
\begin{importanttheorem}[Existence and uniqueness of eigenvector centrality]
Let $\G = (V, E)$ be a strongly connected digraph (or connected undirected graph) with adjacency matrix $A \in \M_n$. Then $A$ is irreducible and non-negative, so by the Perron-Frobenius theorem (Unit 08):
\begin{enumerate}
\item The spectral radius $\lambda_1 = \rho(A) > 0$ is a simple eigenvalue of $A$
\item There exists a unique eigenvector $\bx > \b0$ (the \defword{Perron eigenvector}) satisfying
\[
A \bx = \rho(A)\bx \quad \text{and} \quad \11_n^T \bx = \sum_{i=1}^n x_i = 1
\]
\item No other eigenvalue has a non-negative eigenvector
\end{enumerate}
\end{importanttheorem}
\end{frame}

\begin{frame}{Constructing a stochastic matrix from an adjacency matrix}
Let $A = [a_{ij}] \in \M_n$ be the adjacency matrix of a simple undirected graph $\G = (V, E)$ with no isolated vertices, and let $D = \diag(d_{11}, \ldots, d_{nn})$ be its degree matrix, with diagonal entries
\[
d_{ii} = \sum_{j=1}^n a_{ji} = \sum_{j=1}^n a_{ij} > 0
\]
\vfill
\begin{proposition}[Random walk transition matrix on an undirected graph]
\label{prop:ADinv-column-stochastic}
The matrix $P = A D^{-1}$ is \textbf{column-stochastic}, and defines the transition matrix of a random walk on $\G$ where a walker at vertex $j$ moves to a uniformly chosen neighbour $i \in \N(j)$ with probability $1/d_{jj}$
\end{proposition}
\end{frame}

\begin{frame}[red]{Proof of Proposition~\ref{prop:ADinv-column-stochastic}}
Write the entries of the diagonal matrix $D^{-1}$ as $d_{ij}^{-1}$, where $d_{ii}^{-1} = 1/d_{ii}$ for $1 \le i \le n$ and $d_{ij}^{-1} = 0$ for $i \neq j$
\vfill
By matrix multiplication, the $(i, j)$ entry of $A D^{-1}$ is
\[
(A D^{-1})_{ij} = \sum_{k=1}^n a_{ik} d_{kj}^{-1} = a_{ij} d_{jj}^{-1} = \frac{a_{ij}}{d_{jj}} \ge 0, \qquad i, j = 1, \ldots, n
\]
\vfill
Therefore, the sum of column $j$ of $A D^{-1}$ is
\begin{align*}
\sum_{i=1}^n a_{ij} d_{jj}^{-1}
&= d_{jj}^{-1} \sum_{i=1}^n a_{ij} \\
&= d_{jj}^{-1} d_{jj} = 1
\end{align*}
Hence every column sums to $1$ and $A D^{-1}$ is column-stochastic \hfill $\blacksquare$
\end{frame}

\begin{frame}{Computing eigenvector centrality -- The power method}
For large networks, computing the full characteristic polynomial is impossible. Instead, we compute the dominant eigenpair $(\lambda_1, \bx)$ using the \defword{power method}:
\vfill
Start with an initial positive vector $\bx^{(0)} > \b0$ (such as $\bx^{(0)} = \frac{1}{n}\11_n$). For $k = 1, 2, \ldots$:
\begin{enumerate}
\item Multiply by $A$: $\by^{(k)} = A \bx^{(k-1)}$
\item Normalise (using the $\ell_1$ norm $\|\by^{(k)}\|_1 = \11_n^T \by^{(k)}$ or the maximum-magnitude entry $m(\by^{(k)})$):
\[
\bx^{(k)} = \frac{\by^{(k)}}{\|\by^{(k)}\|_1}
\]
\end{enumerate}
until $\|\bx^{(k)} - \bx^{(k-1)}\|_1 < \varepsilon$
\vfill
\begin{theorem}[Convergence of the power method]
If $A$ is primitive (so $\lambda_1 = \rho(A) > |\lambda_2| \ge \cdots \ge |\lambda_n|$), then $\bx^{(k)} \to \bx$ as $k \to \infty$, and the error decays at the geometric rate
\[
\|\bx^{(k)} - \bx\| = O\!\left(\left|\frac{\lambda_2}{\lambda_1}\right|^k\right)
\]
\end{theorem}
\end{frame}

\SSsubsection{Issue with directed graphs and Katz centrality}

\begin{frame}{What goes wrong on general directed graphs?}
Real-world directed networks (citation graphs, web graphs, DAGs) are rarely strongly connected! When $\G(A)$ is \textbf{reducible}, eigenvector centrality suffers from a severe pathology:
\vfill
\begin{itemize}
\item Suppose vertex $i$ is a \textbf{source node} (in-degree $d_i^{\text{in}} = 0$, so nobody points to $i$)
\item Then row $i$ of $A$ is all zeros, so the eigenvector equation $A\bx = \lambda\bx$ with $\lambda > 0$ gives
\[
\lambda x_i = \sum_{j=1}^n a_{ij} x_j = 0 \implies x_i = 0
\]
\item \textbf{Domino effect:} If vertex $k$ only receives links from source nodes with $x_i = 0$, then $\lambda x_k = \sum_{j \to k} x_j = 0 \implies x_k = 0$ as well!
\end{itemize}
\vfill

Here $x_1 = 0 \implies x_2 = 0 \implies x_3 = 0$, and only vertex $4$ gets non-zero eigenvector centrality ($x_4 = 1$)!
\end{frame}

\begin{frame}{Katz centrality (1953) -- Baseline ``free'' importance}
Leo Katz (1953) resolved the zero-propagation problem on directed graphs by giving every vertex a small baseline ``free'' importance $\beta > 0$, plus damped importance $\alpha > 0$ from its in-neighbours
\vfill
\begin{importantdefinition}[Katz centrality]
Let $A \in \M_n$ be the adjacency matrix of a directed graph, $\alpha > 0$ be an \defword{attenuation factor} (or damping parameter) and $\beta > 0$ be a baseline constant. The \defword{Katz centrality} vector $\bx \in \IR^n$ is the solution to
\[
x_i = \alpha \sum_{j=1}^n a_{ij} x_j + \beta, \quad i=1,\ldots,n
\iff
\bx = \alpha A \bx + \beta \11_n
\]
Rearranging terms gives the linear system
\[
(\II_n - \alpha A)\bx = \beta \11_n
\]
\end{importantdefinition}
\vfill
Since $\beta$ only scales $\bx$ by a positive constant, we typically set $\beta = 1$ (or normalise $\bx$ afterwards so that $\11_n^T \bx = 1$)
\end{frame}

\begin{frame}{Neumann series and walk-counting interpretation of Katz centrality}
When is $\II_n - \alpha A$ guaranteed to be invertible with a strictly positive solution $\bx > \b0$?
\vfill
\begin{importanttheorem}[Existence and walk-counting expansion of Katz centrality]
Let $A \in \M_n$ be a non-negative adjacency matrix and suppose $0 < \alpha < \frac{1}{\rho(A)}$ and $\beta > 0$. Then $\rho(\alpha A) = \alpha \rho(A) < 1$, so by Theorem~\ref{th:neumann-series} (the Neumann series), $\II_n - \alpha A$ is invertible and
\[
\bx = \beta (\II_n - \alpha A)^{-1} \11_n = \beta \left(\sum_{k=0}^\infty \alpha^k A^k\right) \11_n = \beta \left(\11_n + \alpha A \11_n + \alpha^2 A^2 \11_n + \cdots\right) > \b0
\]
\end{importanttheorem}
\vfill
\textbf{Walk-counting interpretation:}
\begin{itemize}
\item Recall from Unit 07 that $(A^k)_{ij}$ is the number of directed walks of length $k$ from $j$ to $i$
\item Therefore $(A^k \11_n)_i = \sum_{j=1}^n (A^k)_{ij}$ is the total number of directed walks of length $k$ from \emph{any} vertex $j$ terminating at vertex $i$
\item Katz centrality sums walks of \textbf{all lengths} $k \ge 0$ ending at $i$, discounted by $\alpha^k$:
\[
x_i = \beta \big(\underbrace{1}_{k=0} + \alpha \underbrace{d_i^{\text{in}}}_{k=1} + \alpha^2 \underbrace{(\text{\# walks of length } 2 \text{ into } i)}_{k=2} + \cdots\big)
\]
\end{itemize}
\end{frame}

\SSsection{PageRank and the Google matrix}

\SSsubsection{Motivation and vote splitting}

\begin{frame}{What makes an important webpage?}
In the early days of the World Wide Web (mid-1990s), search engines ranked pages primarily by keyword frequency on the page itself, which was easily manipulated
\vfill
In 1998, Sergey Brin, Larry Page, Rajeev Motwani and Terry Winograd introduced \defword{PageRank}, the algorithm at the heart of Google
\vfill
\textbf{Core idea:} Use the directed hyperlink structure of the Web to measure authority
\begin{itemize}
\item Webpages are vertices $V = \{1, \ldots, n\}$; a hyperlink from page $j$ to page $i$ is a directed arc $j \to i$
\item A link $j \to i$ is a recommendation or ``vote'' from page $j$ for the importance of page $i$
\item \textbf{Random surfer model:} Imagine a web surfer who clicks links at random from page to page. The PageRank of page $i$ is the long-run proportion of time the random surfer spends on page $i$
\end{itemize}
\end{frame}

\begin{frame}{From Katz centrality to vote splitting}
Why not simply use Katz or eigenvector centrality on the Web adjacency matrix $A$?
\vfill
\textbf{The ``generous hub'' flaw of Katz/eigenvector centrality:}
\begin{itemize}
\item In Katz centrality ($x_i = \alpha \sum_{j \to i} x_j + \beta$), if an important directory page $j$ links to $10{,}000$ pages, \emph{every single one} of those $10{,}000$ pages receives $\alpha x_j$ importance!
\item A page with more outgoing links exerts more total influence on the network
\end{itemize}
\vfill
\textbf{Brin and Page's vote-splitting principle:}
\begin{itemize}
\item Each page $j$ has a single unit of voting power equal to its own importance $x_j$
\item If page $j$ has $d_j^{\text{out}} > 0$ outgoing links, it splits its importance \textbf{equally} among them, passing $x_j / d_j^{\text{out}}$ along each outgoing link:
\[
x_i = \sum_{j \to i} \frac{x_j}{d_j^{\text{out}}}
\]
\end{itemize}
\end{frame}

\begin{frame}{The hyperlink transition matrix}
Let $A = [a_{ij}] \in \M_n$ be the directed adjacency matrix ($a_{ij} = 1$ if page $j$ links to page $i$, with no self-loops). Suppose first that every page has at least one outgoing link ($d_j^{\text{out}} = \sum_{i=1}^n a_{ij} \ge 1$)
\vfill
Define the diagonal out-degree matrix $D^{\text{out}} = \diag(d_1^{\text{out}}, \ldots, d_n^{\text{out}})$ and the \defword{hyperlink matrix} $P = [p_{ij}] \in \M_n$:
\[
p_{ij} = \begin{cases}
\dfrac{1}{d_j^{\text{out}}} & \text{if page } j \text{ links to page } i \\
0 & \text{otherwise}
\end{cases}
\qquad \Longleftrightarrow \qquad
P = A (D^{\text{out}})^{-1}
\]
\vfill
\begin{proposition}
If $d_j^{\text{out}} \ge 1$ for all $j = 1, \ldots, n$, then $P = A(D^{\text{out}})^{-1}$ is \textbf{column-stochastic}, and the vote-splitting equations $x_i = \sum_{j \to i} x_j / d_j^{\text{out}}$ take the fixed-point form
\[
P \bx = \bx
\]
\end{proposition}
\end{frame}


%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%
\SSsubsection{Obstacles on real web graphs and the Google matrix}

\begin{frame}{Two linear algebra obstacles on real web graphs}
If we try to solve $P\bx = \bx$ directly on the real Web, two fatal problems occur:
\vfill
\begin{columns}[T]
\begin{column}{0.48\textwidth}
\textbf{Obstacle 1: Dangling nodes ($d_j^{\text{out}} = 0$)}
\begin{itemize}
\item Many pages (PDF files, images, dead-end pages) have \textbf{no outgoing links}: $d_j^{\text{out}} = 0$
\item Column $j$ of the raw hyperlink matrix is $\b0$, so probability mass ``leaks'' out of the network and $P$ is not stochastic!
\end{itemize}

\textbf{Obstacle 2: Reducibility and spider traps}
\begin{itemize}
\item The Web is not strongly connected ($\G(P)$ is reducible) and contains \textbf{spider traps} (closed strongly connected components) and periodic cycles
\item $P\bx = \bx$ may have multiple linearly independent solutions, assign $x_i = 0$ outside traps, or cause power iteration to oscillate!
\end{itemize}

\begin{frame}{Fixing Obstacle 1 -- Handling dangling nodes}
Before forming the Google matrix, we must ensure that $P$ is genuinely column-stochastic ($\11_n^T P = \11_n^T$)
\vfill
Whenever a vertex $j$ is a \defword{dangling node} ($d_j^{\text{out}} = 0$, giving a zero column in the raw hyperlink matrix), there are two standard conventions:
\begin{enumerate}
\item \textbf{Uniform teleportation from dead ends:} Replace the zero column $j$ with the uniform probability vector $\frac{1}{n}\11_n = (1/n, \ldots, 1/n)^T$ (a random surfer reaching a dead-end page picks any page on the Web uniformly at random)
\item \textbf{Self-loop at dead ends:} Add a self-loop $j \to j$ so that $p_{jj} = 1$ (column $j$ becomes $\be_j$)
\end{enumerate}
\vfill
In both conventions, every column of the resulting matrix $P$ is a valid probability vector ($P \ge 0$ and $\11_n^T P = \11_n^T$). From now on, $P \in \M_n$ denotes this column-stochastic web transition matrix
\end{frame}

\begin{frame}{Fixing Obstacle 2 -- Teleportation and the Google matrix}
To overcome reducibility, spider traps and periodicity, Brin and Page introduced a \defword{damping factor} (or teleportation parameter) $\alpha \in (0, 1)$, typically chosen as $\alpha = 0.85$:
\begin{itemize}
\item With probability $\alpha$, the random surfer follows an outgoing link on the current page (according to $P$)
\item With probability $1 - \alpha$, the surfer gets bored and jumps (teleports) to a page chosen uniformly at random from all $n$ pages (according to $\frac{1}{n}\11_n$)
\end{itemize}
\vfill
\begin{importantdefinition}[Google matrix]
Let $P \in \M_n$ be a column-stochastic transition matrix and let $\alpha \in (0, 1)$ be a damping factor. The \defword{Google matrix} is the $n \times n$ matrix
\[
G = \alpha P + (1 - \alpha)\frac{1}{n}\11_n \11_n^T = \alpha P + \frac{1 - \alpha}{n}\IJ_n
\]
where $\IJ_n = \11_n \11_n^T \in \M_n$ is the $n \times n$ matrix of all ones
\end{importantdefinition}
\end{frame}

\begin{frame}{Existence, uniqueness and convergence of PageRank}
\begin{importanttheorem}[PageRank theorem]
\label{th:pagerank-theorem}
Let $P \in \M_n$ be a column-stochastic matrix and let $\alpha \in (0, 1)$. Then the Google matrix
\[
G = \alpha P + (1-\alpha)\frac{1}{n}\11_n \11_n^T
\]
is \textbf{column-stochastic} and \textbf{strictly positive} ($G > 0$), hence \textbf{primitive}. By Perron's theorem (Unit 08):
\begin{enumerate}
\item The spectral radius $\lambda_1 = \rho(G) = 1$ is a simple, strictly dominant eigenvalue of $G$ (in fact, $|\lambda| \le \alpha < 1$ for every other eigenvalue of $G$)
\item There exists a unique stationary probability vector $\bx > \b0$ (the \defword{PageRank vector}) satisfying
\[
G \bx = \bx \quad \text{and} \quad \11_n^T \bx = 1
\]
\item For any initial probability distribution $\bx^{(0)}$, the power iteration $\bx^{(k)} = G^k \bx^{(0)}$ converges to $\bx$ as $k \to \infty$ at the geometric rate $O(\alpha^k)$
\end{enumerate}
\end{importanttheorem}
\end{frame}

\begin{frame}[red]{Proof of Theorem~\ref{th:pagerank-theorem}}
\textbf{1. Strict positivity ($G > 0$):} Since $P \ge 0$ and $\alpha \in (0, 1)$, we have $\alpha P \ge 0$. Every entry of $\frac{1-\alpha}{n}\IJ_n$ equals $\frac{1-\alpha}{n} > 0$. Therefore, every entry of $G$ satisfies
\[
g_{ij} = \alpha p_{ij} + \frac{1-\alpha}{n} \ge \frac{1-\alpha}{n} > 0
\]
so $G > 0$ is strictly positive (and therefore irreducible and primitive, as $G^1 > 0$)
\vfill
\textbf{2. Column-stochasticity ($\11_n^T G = \11_n^T$):} Since $\11_n^T P = \11_n^T$ and $\11_n^T \11_n = n$, we compute
\[
\11_n^T G = \alpha (\11_n^T P) + \frac{1-\alpha}{n} (\11_n^T \11_n) \11_n^T = \alpha \11_n^T + \frac{1-\alpha}{n} n \11_n^T = (\alpha + 1 - \alpha)\11_n^T = \11_n^T
\]
Thus $G$ is column-stochastic, which implies $\rho(G) = 1$
\vfill
\textbf{3. Application of Perron's theorem:} Since $G > 0$ is a positive column-stochastic matrix, Perron's theorem guarantees that $\lambda_1 = \rho(G) = 1$ is simple and strictly dominant, with a unique positive eigenvector $\bx > \b0$ normalised by $\11_n^T \bx = 1$, and $\lim_{k\to\infty} G^k \bx^{(0)} = \bx$ \hfill $\blacksquare$
\end{frame}

\SSsubsection{Linear system formulation and examples}

\begin{frame}{Reformulating PageRank as a linear system}
Although $G\bx = \bx$ is an eigenvalue problem for the eigenvalue $\lambda = 1$, we can use the normalisation condition $\11_n^T \bx = 1$ to turn it into a \textbf{non-homogeneous linear system} $(\II_n - \alpha P)\bx = \frac{1-\alpha}{n}\11_n$!
\vfill
\begin{importanttheorem}[Linear system formulation of PageRank]
\label{th:pagerank-linear-system}
Let $P \in \M_n$ be column-stochastic and $\alpha \in (0, 1)$. A vector $\bx \in \IR^n$ with $\11_n^T \bx = 1$ satisfies $G\bx = \bx$ if and only if it satisfies the linear system
\[
(\II_n - \alpha P)\bx = \frac{1 - \alpha}{n}\11_n
\]
Moreover, $\II_n - \alpha P$ is invertible and its inverse is given by the Neumann series $(\II_n - \alpha P)^{-1} = \sum_{k=0}^\infty \alpha^k P^k$, so the unique PageRank vector is
\[
\bx = \frac{1 - \alpha}{n} (\II_n - \alpha P)^{-1} \11_n
\]
\end{importanttheorem}
\end{frame}

\begin{frame}[red]{Proof of Theorem~\ref{th:pagerank-linear-system}}
\textbf{1. Equivalence with $G\bx = \bx$:} For any vector $\bx$ satisfying $\11_n^T \bx = 1$, associativity of matrix multiplication gives
\[
\IJ_n \bx = (\11_n \11_n^T)\bx = \11_n (\11_n^T \bx) = \11_n (1) = \11_n
\]
Substituting this into $G\bx = \bx$ gives
\[
G\bx = \bx \iff \alpha P \bx + \frac{1-\alpha}{n}\11_n \11_n^T \bx = \bx \iff \alpha P \bx + \frac{1-\alpha}{n}\11_n = \bx \iff (\II_n - \alpha P)\bx = \frac{1-\alpha}{n}\11_n
\]
\textbf{2. Invertibility of $\II_n - \alpha P$:} Since $P$ is column-stochastic, $\rho(P) = 1$. Because $\alpha \in (0, 1)$, we have $\rho(\alpha P) = \alpha \rho(P) = \alpha < 1$. By Theorem~\ref{th:neumann-series} (the Neumann series), $\II_n - \alpha P$ is invertible with $(\II_n - \alpha P)^{-1} = \sum_{k=0}^\infty \alpha^k P^k \ge \b0$
\vfill
\textbf{3. Automatic normalisation ($\11_n^T \bx = 1$):} Left-multiplying $(\II_n - \alpha P)\bx = \frac{1-\alpha}{n}\11_n$ by $\11_n^T$ and using $\11_n^T P = \11_n^T$ yields
\[
\11_n^T (\II_n - \alpha P)\bx = (1 - \alpha)\11_n^T \bx = \frac{1-\alpha}{n}\11_n^T \11_n = 1 - \alpha \implies \11_n^T \bx = 1 \qquad \blacksquare
\]
\end{frame}

\begin{frame}{Unifying thread of Unit 09 -- The resolvent $(\II - M)^{-1}$}
Notice how all three topics in this unit are governed by the exact same matrix inverse (the \textbf{Neumann series}):
\vfill
\begin{center}
\renewcommand{\arraystretch}{1.6}
\begin{tabular}{l|c|c|c}
\textbf{Problem} & \textbf{Matrix $M$} & \textbf{Condition $\rho(M) < 1$} & \textbf{Key quantity} \\ \hline
Absorbing Markov chains & Transient block $Q$ & Absorption reachable & $N = (\II_m - Q)^{-1}$ \\
Katz centrality & $\alpha A$ & $0 < \alpha < 1/\rho(A)$ & $\bx = \beta(\II_n - \alpha A)^{-1}\11_n$ \\
PageRank & $\alpha P$ & $0 < \alpha < 1$ ($\rho(P)=1$) & $\bx = \frac{1-\alpha}{n}(\II_n - \alpha P)^{-1}\11_n$
\end{tabular}
\end{center}
\vfill
For hand calculations on small graphs ($n = 3$ or $4$), solving the linear system $(\II_n - \alpha P)\bx = \frac{1-\alpha}{n}\11_n$ is much faster than computing the dense matrix $G$ and finding its nullspace!
\end{frame}

\textbf{Substitute into the normalisation condition $x_1 + x_2 + x_3 = 1$:}
\[
x_1 + \left(\frac{1}{4}x_1 + \frac{1}{6}\right) + \left(2x_1 - \frac{1}{3}\right) = 1
\implies \frac{13}{4}x_1 = \frac{7}{6}
\implies x_1 = \frac{14}{39}
\]
Then $x_2 = \frac{1}{4}\left(\frac{14}{39}\right) + \frac{1}{6} = \frac{10}{39}$ and $x_3 = 2\left(\frac{14}{39}\right) - \frac{13}{39} = \frac{15}{39}$, so
\[
\bx = \begin{pmatrix} 14/39 \\ 10/39 \\ 15/39 \end{pmatrix} \approx \begin{pmatrix} 0.359 \\ 0.256 \\ 0.385 \end{pmatrix} \quad \implies \quad \text{Ranking: Page } 3 \succ \text{Page } 1 \succ \text{Page } 2
\]
\end{frame}

\begin{frame}{Why Page 1 ranks almost as high as Page 3}
Look closely at the ranking in our 3-page example:
\begin{itemize}
\item \textbf{In-degrees:} Page $3$ has in-degree $2$ (from $1$ and $2$), while Page $1$ and Page $2$ both have in-degree $1$
\item By raw in-degree, Page $1$ and Page $2$ would be tied!
\item However, in PageRank, Page $1$ receives the \textbf{undivided vote} of the most important page (Page $3$, with $d_3^{\text{out}} = 1$), whereas Page $2$ only receives \textbf{half a vote} from Page $1$ ($d_1^{\text{out}} = 2$)
\item Consequently, Page $1$ ($x_1 = 14/39$) ranks far above Page $2$ ($x_2 = 10/39$) and nearly matches Page $3$ ($x_3 = 15/39$)
\end{itemize}
\vfill
What if we use the standard Google damping factor $\alpha = 0.85$ on the same 3-page network?
\[
(\II_3 - 0.85 P)\bx = \frac{0.15}{3}\11_3 \implies \bx \approx \begin{pmatrix} 0.388 \\ 0.215 \\ 0.397 \end{pmatrix}
\]
\end{frame}

\begin{frame}{Worked exam-style example 2 -- 4-page web with a dangling node}
Suppose we add a dangling Page $4$ ($d_4^{\text{out}} = 0$) to a 4-page web where $1 \to 2$, $1 \to 3$, $2 \to 3$, $3 \to 1$ and $2 \to 4$ (so $d_1^{\text{out}} = 2$, $d_2^{\text{out}} = 2$, $d_3^{\text{out}} = 1$ and $d_4^{\text{out}} = 0$):
\vfill
\[
P_{\text{raw}} = \begin{pmatrix}
0 & 0 & 1 & 0 \\
1/2 & 0 & 0 & 0 \\
1/2 & 1/2 & 0 & 0 \\
0 & 1/2 & 0 & 0
\end{pmatrix}
\quad \xrightarrow{\text{replace zero col 4 by }\frac{1}{4}\11_4} \quad
P = \begin{pmatrix}
0 & 0 & 1 & 1/4 \\
1/2 & 0 & 0 & 1/4 \\
1/2 & 1/2 & 0 & 1/4 \\
0 & 1/2 & 0 & 1/4
\end{pmatrix}
\]
\vfill
Notice why we can simplify $(\II_4 - \alpha P)\bx = \frac{1-\alpha}{4}\11_4$ even faster when column $4$ is $\frac{1}{4}\11_4$:
\begin{itemize}
\item Write $P = P_{\text{raw}} + \frac{1}{4}\11_4 \be_4^T$, so $P\bx = P_{\text{raw}}\bx + \frac{x_4}{4}\11_4$
\item Then $G\bx = \bx \iff (\II_4 - \alpha P_{\text{raw}})\bx = \left(\frac{1-\alpha}{4} + \frac{\alpha x_4}{4}\right)\11_4 = c \11_4$, where $P_{\text{raw}}$ is sparse!
\item Thus we can solve $(\II_4 - \alpha P_{\text{raw}})\by = \11_4$ by sparse back-substitution and normalise $\bx = \by / (\11_4^T \by)$!
\end{itemize}
\end{frame}

\begin{frame}{Computing PageRank at web scale -- Sparse power iteration}
For the real Web ($n \sim 10^{10}$ pages), how is PageRank computed in practice?
\vfill
\begin{itemize}
\item \textbf{Direct Gaussian elimination} on $(\II_n - \alpha P)\bx = \frac{1-\alpha}{n}\11_n$ costs $O(n^3)$ time and $O(n^2)$ memory due to fill-in, which is completely impossible for $n = 10^{10}$
\item \textbf{Forming the dense Google matrix $G$} explicitly also requires $n^2$ entries because $\IJ_n$ is completely dense!
\item \textbf{Matrix-free power iteration:} Starting from $\bx^{(0)} = \frac{1}{n}\11_n$, for any probability vector $\bx^{(k)}$ ($\11_n^T \bx^{(k)} = 1$) we compute
\[
\bx^{(k+1)} = G \bx^{(k)} = \alpha P \bx^{(k)} + \frac{1 - \alpha}{n}\11_n
\]
\end{itemize}
\vfill
\textbf{Why is this fast?}
\begin{itemize}
\item We \textbf{never} form $G$: each step only requires one sparse matrix-vector multiplication $P \bx^{(k)}$ ($O(m)$ operations where $m \approx 10n$ is the number of hyperlinks) plus adding the scalar $\frac{1-\alpha}{n}$ to each entry!
\item Since $|\lambda_2(G)| \le \alpha = 0.85$, after $k = 50$ iterations the error is bounded by $0.85^{50} \approx 3 \times 10^{-4}$
\end{itemize}
\end{frame}

\begin{frame}{Summary of Unit 09}
\begin{itemize}
\item \textbf{Absorbing Markov chains:} Reordering states so the $r$ absorbing states precede the $m$ transient states gives the canonical column-stochastic block form
\[
P = \begin{pmatrix} \II_r & R \\ \b0_{m,r} & Q \end{pmatrix}, \qquad \rho(Q) < 1
\]
\item \textbf{Fundamental matrix:} $N = (\II_m - Q)^{-1} = \sum_{k=0}^\infty Q^k$ counts expected visits to transient states, $\bt^T = \11_m^T N$ gives expected absorption times and $B = R N$ gives absorption probabilities
\item \textbf{Spectral centrality:} Eigenvector centrality ($A\bx = \lambda\bx$) requires strong connectivity; Katz centrality ($(\II_n - \alpha A)\bx = \beta\11_n$ for $\alpha < 1/\rho(A)$) fixes directed acyclic and source nodes by adding baseline importance
\item \textbf{PageRank:} Normalising votes by out-degree and adding teleportation yields the primitive column-stochastic Google matrix $G = \alpha P + \frac{1-\alpha}{n}\11_n\11_n^T$, whose unique stationary vector solves $(\II_n - \alpha P)\bx = \frac{1-\alpha}{n}\11_n$
\end{itemize}
\end{frame}
