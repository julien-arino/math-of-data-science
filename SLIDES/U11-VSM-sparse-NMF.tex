\begin{frame}{Why represent text using linear algebra?}
\begin{itemize}
\item Much of the data generated in science, industry and daily life consists of unstructured text: books, articles, clinical notes and web pages
\vfill
\item Computers cannot perform calculus or geometry directly on strings of characters
\vfill
\item In this unit, we turn discrete text corpora into \defword{sparse non-negative data matrices} and apply three core tools of linear algebra:
\begin{enumerate}
\item \textbf{Diagonal matrix scaling} (TF-IDF) to normalise rows (documents) and reweight columns (terms)
\item \textbf{Inner products and angles} (cosine similarity) to measure semantic similarity in high dimensions
\item \textbf{Constrained low-rank factorisation} (Non-negative Matrix Factorisation via Alternating Least Squares) to discover latent topics
\end{enumerate}
\end{itemize}
\end{frame}

\SSsection{Text tokenisation and the vector space model}

\begin{frame}{The central idea: from words to vectors}
\begin{itemize}
\item The fundamental challenge in natural language processing (\defword{NLP}) and information retrieval is to translate discrete symbolic data (\defword{words} and \defword{documents}) into a numerical format
\vfill
\item Why? Mathematical models are built on vector spaces, inner products and matrix factorisations rather than dictionary definitions
\vfill
\item The \defword{vector space model} (\defword{VSM}), introduced in information retrieval by Salton, Wong and Yang (1975), represents documents and words as vectors in $\IR^p$ so that the geometry of the space reflects semantic relationships
\vfill
\item Documents or words with similar meaning are mapped to vectors pointing in similar directions in $\IR^p$
\end{itemize}
\end{frame}

\begin{frame}{What is a token?}
\begin{itemize}
\item A \defword{token} is the fundamental discrete unit of text processed by a model: it can be a word, a word stem, a sub-word or a character
\vfill
\item Converting a raw character string into a sequence of tokens is called \defword{tokenisation}
\vfill
\item For example, after lower-casing and punctuation removal, the sentence ``The cat sat on the mat'' is tokenised as
\[
[\code{"the"},\; \code{"cat"},\; \code{"sat"},\; \code{"on"},\; \code{"the"},\; \code{"mat"}]
\]
\vfill
\item Each unique token in our collection is assigned an integer feature index $j \in \{1,\ldots,p\}$ from a fixed vocabulary, yielding an integer sequence such as $[5, 1, 4, 3, 5, 2]$
\end{itemize}
\end{frame}


\begin{frame}{Vocabulary and standard basis (one-hot) vectors}
\begin{importantdefinition}[Vocabulary and tokenisation]
A \defword{vocabulary} $V = \{w_1, w_2, \ldots, w_p\}$ is an ordered set of $p = |V|$ distinct tokens (serving as the $p$ features). \defword{Tokenisation} maps a raw document $d$ to a finite sequence of tokens $(t_1, t_2, \ldots, t_L)$ with each $t_\ell \in V$
\end{importantdefinition}
\vfill
\begin{importantdefinition}[One-hot representation of a token]
Given a vocabulary $V = \{w_1, \ldots, w_p\}$ of size $p$, the \defword{one-hot vector} associated with token $w_j \in V$ is the $j$-th standard basis vector $\be_j \in \IR^p$, whose $k$-th entry is
\[
(\be_j)_k = \delta_{jk} =
\begin{cases}
1 & \textrm{if } k = j \\
0 & \textrm{if } k \neq j
\end{cases}
\]
\end{importantdefinition}
\end{frame}

\begin{frame}{From tokens to documents: the bag-of-words model}
How do we represent a whole document $d_i$ ($i \in \{1, \ldots, n\}$) as a single feature vector in $\IR^p$?
\vfill
\begin{importantdefinition}[Bag-of-words document vector]
Let $V = \{w_1, \ldots, w_p\}$ be a vocabulary of size $p$ and let $d_i$ be a tokenised document $(t_1, t_2, \ldots, t_{L_i})$. The \defword{bag-of-words} (\defword{BoW}) vector of document $d_i$ is
\[
\bx_i = \sum_{\ell=1}^{L_i} \be_{\mathsf{idx}(t_\ell)} = \sum_{j=1}^p f_{ij} \be_j =
\begin{pmatrix}
f_{i1} \\ f_{i2} \\ \vdots \\ f_{ip}
\end{pmatrix} \in \IRplus^p
\]
where $f_{ij} \in \IN_0$ is the number of times term $w_j$ occurs in document $d_i$
\end{importantdefinition}
\end{frame}


\begin{frame}{Example -- Constructing bag-of-words vectors}
Suppose our vocabulary has $p = 5$ terms:
\[
V = \{w_1=\code{"cat"},\; w_2=\code{"dog"},\; w_3=\code{"mat"},\; w_4=\code{"sat"},\; w_5=\code{"the"}\}
\]
\vfill
Consider two short documents (after lower-casing):
\begin{itemize}
\item $d_1$: \code{"the cat sat on the mat"} (ignoring \code{"on"} $\notin V$)
\item $d_2$: \code{"the dog sat on the dog mat"}
\end{itemize}
\vfill
Summing the standard basis vectors gives their bag-of-words vectors in $\IRplus^5$:
\[
\bx_1 = 1\be_1 + 0\be_2 + 1\be_3 + 1\be_4 + 2\be_5 = \begin{pmatrix} 1 \\ 0 \\ 1 \\ 1 \\ 2 \end{pmatrix}, \qquad
\bx_2 = 0\be_1 + 2\be_2 + 1\be_3 + 1\be_4 + 2\be_5 = \begin{pmatrix} 0 \\ 2 \\ 1 \\ 1 \\ 2 \end{pmatrix}
\]
\end{frame}


\begin{frame}{What bag-of-words preserves and what it discards}
\begin{itemize}
\item Because vector addition in $\IR^p$ is commutative ($\be_j + \be_k = \be_k + \be_j$), the bag-of-words map completely discards \textbf{word order} and grammar
\vfill
\item For example, ``dog bites man'' and ``man bites dog'' produce the exact same bag-of-words vector $\bx \in \IRplus^p$
\vfill
\item Despite this simplification, bag-of-words vectors are remarkably effective for \textbf{topic classification}, \textbf{document clustering} and \textbf{search}:
\begin{itemize}
\item Documents about linear algebra frequently contain \code{"matrix"}, \code{"rank"} and \code{"eigenvalue"}
\item Documents about graph theory frequently contain \code{"vertex"}, \code{"edge"} and \code{"path"}
\end{itemize}
\end{itemize}
\end{frame}


\begin{frame}{Sparsity and non-negativity in high dimensions}
\begin{definition}[Support and number of non-zero entries]
For a vector $\bx = (x_1, \ldots, x_p)^T \in \IR^p$, its \defword{support} is the index set of non-zero entries
\[
\mathsf{supp}(\bx) = \{j \in \{1, \ldots, p\} : x_j \neq 0\}
\]
and we write $\|\bx\|_0 = |\mathsf{supp}(\bx)|$ for the number of non-zero entries of $\bx$
\end{definition}
\vfill
\begin{itemize}
\item Note that $\|\bx\|_0$ is \emph{not} a true norm (since $\|\alpha \bx\|_0 = \|\bx\|_0$ for $\alpha \neq 0$), but it measures \defword{sparsity}
\vfill
\item In realistic text corpora, the vocabulary size $p$ is large ($10^4$ to $10^5$ terms), whereas a single document uses only a few hundred distinct words
\vfill
\item Thus $\|\bx_i\|_0 \ll p$: every document vector $\bx_i \in \IRplus^p$ is a \textbf{high-dimensional, sparse, non-negative vector} ($98\%$ to $99.9\%$ of its entries are zero)
\end{itemize}
\end{frame}


\begin{frame}{Vocabulary pre-processing as feature selection}
In practice, before forming document vectors $\bx_i \in \IRplus^p$, raw tokens undergo two standard pre-processing steps:
\vfill
\begin{enumerate}
\item \textbf{Stop-word removal}: discarding high-frequency function words (\code{"the"}, \code{"and"}, \code{"of"}, \code{"to"}) that carry little topical meaning
\item \textbf{Stemming or lemmatisation}: collapsing inflected forms (\code{"matrices"}, \code{"matrix"} $\to$ \code{"matrix"}; \code{"computing"}, \code{"computed"} $\to$ \code{"comput"}) into a single vocabulary feature
\end{enumerate}
\vfill
\begin{itemize}
\item \textbf{Linear algebra view}: Removing $p - p'$ stop words from a vocabulary of size $p$ is equivalent to left-multiplying each document vector $\bx_i \in \IRplus^p$ by a \textbf{feature-selection matrix} $P \in \M_{p', p}(\{0, 1\})$ obtained by deleting the stop-word rows of $\II_p$:
\[
\bx_i^{\text{reduced}} = P \bx_i \in \IRplus^{p'} \quad\Longleftrightarrow\quad (\bx_i^{\text{reduced}})^T = \bx_i^T P^T \in \IRplus^{1 \times p'}
\]
\end{itemize}
\end{frame}

\SSsubsection{The document-term data matrix}

\begin{frame}{The document-term data matrix}
\begin{importantdefinition}[Document-term data matrix]
Given a corpus $D = \{d_1, \ldots, d_n\}$ of $n$ documents (observations) and a vocabulary $V = \{w_1, \ldots, w_p\}$ of $p$ terms (features), the \defword{data matrix} (called in NLP and information retrieval the \defword{document-term matrix}) is the non-negative matrix
\[
X = [x_{ij}] = \begin{pmatrix} \bx_1^T \\ \bx_2^T \\ \vdots \\ \bx_n^T \end{pmatrix} = \begin{bmatrix} \bc_1 & \bc_2 & \cdots & \bc_p \end{bmatrix} \in \M_{np}(\IRplus)
\]
where:
\begin{itemize}
\item Row $i \in \{1, \ldots, n\}$ is the row vector $\bx_i^T \in \IRplus^{1 \times p}$ of document (observation) $d_i$
\item Column $j \in \{1, \ldots, p\}$ is the column vector $\bc_j \in \IRplus^n$ of vocabulary term (feature) $w_j$
\item Entry $x_{ij} \ge 0$ is the weight (e.g., raw frequency $f_{ij}$) of term $w_j$ in document $d_i$
\end{itemize}
\end{importantdefinition}
\vfill
\begin{itemize}
\item Some authors work with the transpose $X^T \in \M_{pn}(\IRplus)$, called the \defword{term-document matrix}; we follow our standard data matrix convention with observations (documents) in rows and features (terms) in columns
\end{itemize}
\end{frame}


\begin{frame}{Example -- A toy corpus of $n=4$ documents and $p=6$ terms}
Consider $n=4$ short documents across two mathematical themes, tracked over the vocabulary $V = \{\code{matrix}, \code{vector}, \code{eigenvalue}, \code{graph}, \code{network}, \code{data}\}$ ($p=6$):
\vfill
\begin{itemize}
\item \textbf{Document $d_1$ (linear algebra)}: mentions \code{matrix} (3 times), \code{vector} (2 times), \code{eigenvalue} (2 times) and \code{data} (4 times)
\vfill
\item \textbf{Document $d_2$ (linear algebra)}: mentions \code{matrix} (2 times), \code{vector} (3 times) and \code{data} (3 times)
\vfill
\item \textbf{Document $d_3$ (graph theory)}: mentions \code{graph} (3 times), \code{network} (2 times) and \code{data} (5 times)
\vfill
\item \textbf{Document $d_4$ (graph theory)}: mentions \code{vector} (1 time), \code{graph} (2 times), \code{network} (3 times) and \code{data} (4 times)
\end{itemize}
\end{frame}


\begin{frame}{Example -- Explicit $4 \times 6$ document-term matrix $X$}
Assembling the document vectors $\bx_1^T, \bx_2^T, \bx_3^T, \bx_4^T \in \IRplus^{1 \times 6}$ as rows gives $X \in \M_{46}(\IRplus)$:
\[
\begin{array}{r|cccccc}
\textrm{Doc} & \code{matrix} & \code{vector} & \code{eigen} & \code{graph} & \code{network} & \code{data} \\
\hline
d_1 & 3 & 2 & 2 & 0 & 0 & 4 \\
d_2 & 2 & 3 & 0 & 0 & 0 & 3 \\
d_3 & 0 & 0 & 0 & 3 & 2 & 5 \\
d_4 & 0 & 1 & 0 & 2 & 3 & 4
\end{array}
\implies
X = \begin{pmatrix}
3 & 2 & 2 & 0 & 0 & 4 \\
2 & 3 & 0 & 0 & 0 & 3 \\
0 & 0 & 0 & 3 & 2 & 5 \\
0 & 1 & 0 & 2 & 3 & 4
\end{pmatrix}
\]
\vfill
\begin{itemize}
\item Notice the near-block structure in columns $1$--$5$ separating $\{d_1, d_2\}$ from $\{d_3, d_4\}$
\item In contrast, column $6$ (\code{data}) has large entries across \textbf{all four rows}, obscuring the two topics!
\end{itemize}
\end{frame}


\begin{frame}{Dual geometric views of $X \in \M_{np}(\IRplus)$}
Every document-term data matrix $X \in \M_{np}(\IRplus)$ admits two complementary geometric interpretations:
\vfill
\begin{columns}[t]
\begin{column}{0.48\textwidth}
\textbf{1. Row view (document space $\IRplus^p$)}
\begin{itemize}
\item Each transposed row $\bx_i \in \IRplus^p$ ($i=1,\ldots,n$) represents a \textbf{document} (observation) as a point in $p$-dimensional term space
\item Used to compare, cluster or classify documents
\end{itemize}
\end{column}
\begin{column}{0.48\textwidth}
\textbf{2. Column view (term space $\IRplus^n$)}
\begin{itemize}
\item Each column $\bc_j \in \IRplus^n$ ($j=1,\ldots,p$) represents a \textbf{term} (feature) via its occurrence profile across the $n$ documents
\item Two words that co-occur in the same documents have similar column vectors
\end{itemize}
\end{column}
\end{columns}
\end{frame}

\SSsubsection{TF-IDF as diagonal matrix multiplication}

\begin{frame}{Why raw term counts fail}
Using raw frequency counts $x_{ij} = f_{ij}$ directly in $X$ suffers from two major defects:
\vfill
\begin{enumerate}
\item \textbf{Common words dominate rare discriminative words}
\begin{itemize}
\item Words like \code{"the"}, \code{"is"} or \code{"data"} appear in almost every document with large counts, dominating Euclidean norms and inner products
\item Rare domain terms like \code{"eigenvalue"} or \code{"isomorphism"} occur fewer times but carry far more information about a document's topic
\end{itemize}
\vfill
\item \textbf{Document length distorts vector magnitude}
\begin{itemize}
\item A 20-page article and a 1-paragraph abstract on the exact same topic have proportional term frequencies, but the longer document has a much larger norm $\|\bx_i\|_2$
\end{itemize}
\end{enumerate}
\end{frame}


\begin{frame}{Term frequency and inverse document frequency (TF-IDF)}
\begin{importantdefinition}[Document frequency, IDF and TF-IDF]
Let $X = [f_{ij}] \in \M_{np}(\IRplus)$ be a raw document-term count matrix for $n$ documents and $p$ terms:
\begin{enumerate}
\item The \defword{term frequency} $\mathsf{tf}_{ij}$ is the raw count $f_{ij}$ (or a scaled variant such as $f_{ij} / \sum_{k=1}^p f_{ik}$)
\item The \defword{document frequency} $\mathsf{df}_j$ of term $w_j$ is the number of documents in which $w_j$ appears:
\[
\mathsf{df}_j = \left|\{i \in \{1, \ldots, n\} : f_{ij} > 0\}\right| = \|\bc_j\|_0 \in \{1, \ldots, n\}
\]
\item The \defword{inverse document frequency} of term $w_j$ is
\[
\mathsf{idf}_j = \log\!\left(\frac{n}{\mathsf{df}_j}\right) \ge 0
\]
\item The \defword{TF-IDF weight} of term $w_j$ in document $d_i$ is $\mathsf{tfidf}_{ij} = \mathsf{tf}_{ij} \cdot \mathsf{idf}_j$
\end{enumerate}
\end{importantdefinition}
\end{frame}


\begin{frame}{How $\mathsf{idf}_j = \log(n / \mathsf{df}_j)$ reweights vocabulary terms}
Assume every vocabulary term $w_j \in V$ appears in at least one document ($1 \le \mathsf{df}_j \le n$):
\vfill
\begin{center}
\begin{tabular}{lccl}
\toprule
\textbf{Term behaviour} & $\mathsf{df}_j$ & $\mathsf{idf}_j = \log(n/\mathsf{df}_j)$ & \textbf{Effect on column $j$ of $X$} \\
\midrule
Appears in all documents & $n$ & $\log(1) = 0$ & Zeroed out completely \\
Appears in half of documents & $n/2$ & $\log(2) \approx 0.693$ & Moderately weighted \\
Appears in 1 document only & $1$ & $\log(n)$ (maximal) & Strongly amplified \\
\bottomrule
\end{tabular}
\end{center}
\vfill
\begin{itemize}
\item Note: throughout this course, $\log(\cdot)$ denotes the natural logarithm $\ln(\cdot)$ (changing the base of the logarithm only multiplies all $\mathsf{idf}_j$ by a positive constant)
\item In practice, when out-of-corpus queries might have $\mathsf{df}_j = 0$, smoothed variants such as $\log\!\left(\frac{n+1}{\mathsf{df}_j+1}\right) + 1$ are also used
\end{itemize}
\end{frame}


\begin{frame}{TF-IDF and length normalisation as diagonal matrix scaling}
\begin{importanttheorem}[Row and column scaling via diagonal matrices]\label{th:tfidf_diag_scaling}
Let $X = [x_{ij}] \in \M_{np}(\IRplus)$ be a document-term data matrix with non-zero rows $\bx_1^T, \ldots, \bx_n^T$. Define the two diagonal matrices
\[
D_{\text{IDF}} = \diag(\mathsf{idf}_1, \ldots, \mathsf{idf}_p) \in \M_p(\IRplus), \qquad
D_{\text{len}}^{-1} = \diag\!\left(\frac{1}{\|\bx_1^{\text{TF-IDF}}\|_2}, \ldots, \frac{1}{\|\bx_n^{\text{TF-IDF}}\|_2}\right) \in \M_n(\IRplus)
\]
where $(\bx_i^{\text{TF-IDF}})^T = \bx_i^T D_{\text{IDF}}$ is the $i$-th row of $X D_{\text{IDF}}$. Then:
\begin{enumerate}
\item \textbf{Right-multiplication} $X_{\text{TF-IDF}} = X D_{\text{IDF}}$ scales the \textbf{columns} (terms/features) of $X$: entry $(i,j)$ is $x_{ij} \, \mathsf{idf}_j$
\item \textbf{Left-multiplication} $D_{\text{len}}^{-1} A$ scales the \textbf{rows} (documents/observations) of $A$: entry $(i,j)$ is $a_{ij} / \|\bx_i^{\text{TF-IDF}}\|_2$
\item The combined matrix $\hat{X} = D_{\text{len}}^{-1} X D_{\text{IDF}} \in \M_{np}(\IRplus)$ has entries $\hat{x}_{ij} = \frac{x_{ij} \, \mathsf{idf}_j}{\|\bx_i^{\text{TF-IDF}}\|_2}$ and unit-norm document rows $\|\hat{\bx}_i\|_2 = 1$
\end{enumerate}
\end{importanttheorem}
\end{frame}


\begin{frame}[red]{Proof of Theorem~\ref{th:tfidf_diag_scaling}}
Let $D_{\text{IDF}} = [c_{kj}] \in \M_p$ with $c_{jj} = \mathsf{idf}_j$ and $c_{kj} = 0$ for $k \neq j$. By definition of matrix multiplication, the $(i,j)$ entry of $X D_{\text{IDF}}$ is
\[
(X D_{\text{IDF}})_{ij} = \sum_{k=1}^p x_{ik} c_{kj} = x_{ij} c_{jj} = x_{ij} \, \mathsf{idf}_j
\]
Thus column $j$ of $X$ is multiplied by the scalar $\mathsf{idf}_j$
\vfill
Similarly, let $D_{\text{len}}^{-1} = [d_{i\ell}] \in \M_n$ with $d_{ii} = 1/\|\bx_i^{\text{TF-IDF}}\|_2$ and $d_{i\ell} = 0$ for $\ell \neq i$. For any matrix $A = [a_{\ell j}] \in \M_{np}$, the $(i,j)$ entry of $D_{\text{len}}^{-1} A$ is
\[
(D_{\text{len}}^{-1} A)_{ij} = \sum_{\ell=1}^n d_{i\ell} a_{\ell j} = d_{ii} a_{ij} = \frac{a_{ij}}{\|\bx_i^{\text{TF-IDF}}\|_2}
\]
Setting $A = X D_{\text{IDF}}$ gives $\hat{x}_{ij} = (D_{\text{len}}^{-1} X D_{\text{IDF}})_{ij} = \frac{x_{ij} \, \mathsf{idf}_j}{\|\bx_i^{\text{TF-IDF}}\|_2}$ \hfill\qed
\end{frame}

\textbf{Step 1: Count non-zero entries in each column to find $\mathsf{df}_j$ and $\mathsf{idf}_j = \log(4/\mathsf{df}_j)$}:

\SSsection{Angles and cosine similarity in high dimensions}

\begin{frame}{Why Euclidean distance fails for sparse document vectors}
Suppose we want to measure how similar two documents $\bx, \by \in \IRplus^p$ are. Why not use the Euclidean distance $\|\bx - \by\|_2$?
\vfill
\begin{itemize}
\item Consider three documents over $V = \{\code{matrix}, \code{vector}, \code{graph}\}$ ($p = 3$):
\[
\bx_1 = \begin{pmatrix} 1 \\ 1 \\ 0 \end{pmatrix}, \qquad
\bx_2 = \begin{pmatrix} 5 \\ 5 \\ 0 \end{pmatrix} = 5\bx_1, \qquad
\bx_3 = \begin{pmatrix} 0 \\ 0 \\ 1 \end{pmatrix}
\]
\item Documents $d_1$ and $d_2$ have the \textbf{exact same word proportions} (pure linear algebra), whereas $d_3$ shares \textbf{no words at all} with $d_1$
\vfill
\item Yet their Euclidean distances are:
\[
\|\bx_1 - \bx_2\|_2 = \left\|\begin{pmatrix} -4 \\ -4 \\ 0 \end{pmatrix}\right\|_2 = \sqrt{32} \approx 5.66
\quad > \quad
\|\bx_1 - \bx_3\|_2 = \left\|\begin{pmatrix} 1 \\ 1 \\ -1 \end{pmatrix}\right\|_2 = \sqrt{3} \approx 1.73
\]
\item Euclidean distance judges $d_1$ to be much closer to the unrelated short document $d_3$ than to a longer copy of itself!
\end{itemize}
\end{frame}


\begin{frame}{Direction matters more than magnitude}
\begin{center}
\begin{tikzpicture}[scale=0.85]
    \draw[->, thick] (0,0) -- (6.5,0) node[right, font=\small\bfseries] {Term 1 (\code{matrix})};
    \draw[->, thick] (0,0) -- (0,4.5) node[above, font=\small\bfseries] {Term 2 (\code{graph})};

    % Vector x1 and x2 = 3 x1

    \draw[->, thick, blue!75!black, dashed] (1.8,0.6) -- (5.4,1.8) node[right, font=\small] {$\bx_2 = 3\bx_1$};

    % Vector x3

    \draw[thick, orange!90!black, ->] (1.1,0.367) arc (18.43:77.2:1.16);

    \draw[dotted, thick, gray!80!black] (1.8,0.6) -- (0.5,2.2);
\end{tikzpicture}

\vfill
\begin{itemize}
\item Scaling a document vector by a positive constant ($\bx_2 = \alpha \bx_1$ with $\alpha > 0$) changes its length $\|\bx\|_2$ but preserves its \textbf{direction}
\item Therefore, semantic similarity between documents should be measured by the \defword{angle} $\theta$ between their vectors
\end{itemize}
\end{frame}


\begin{frame}{Cosine similarity}
\begin{importantdefinition}[Cosine similarity]
Let $\bx, \by \in \IR^p$ be two non-zero vectors. The \defword{cosine similarity} between $\bx$ and $\by$ is the cosine of the angle $\theta \in [0, \pi]$ between them:
\[
\mathsf{sim}(\bx, \by) = \cos(\theta) = \frac{\bx \bullet \by}{\|\bx\|_2 \, \|\by\|_2}
= \frac{\sum_{j=1}^p x_j y_j}{\sqrt{\sum_{j=1}^p x_j^2}\sqrt{\sum_{j=1}^p y_j^2}}
\]
Equivalently, in terms of the $\ell_2$-normalised unit vectors $\hat{\bx} = \frac{\bx}{\|\bx\|_2}$ and $\hat{\by} = \frac{\by}{\|\by\|_2}$,
\[
\mathsf{sim}(\bx, \by) = \hat{\bx} \bullet \hat{\by} = \hat{\bx}^T \hat{\by}
\]
\end{importantdefinition}
\end{frame}


\begin{frame}{Fundamental properties of cosine similarity}
\begin{proposition}[Properties of cosine similarity]\label{prop:cosine_properties}
Let $\bx, \by \in \IR^p$ be non-zero vectors. Then:
\begin{enumerate}
\item \textbf{Positive scale invariance}: For any scalars $\alpha > 0$ and $\beta > 0$,
\[
\mathsf{sim}(\alpha \bx, \beta \by) = \mathsf{sim}(\bx, \by)
\]
\item \textbf{General bounds}: By the Cauchy-Schwarz inequality, $\mathsf{sim}(\bx, \by) \in [-1, 1]$, with $\mathsf{sim}(\bx, \by) = 1 \iff \by = c\bx$ for some $c > 0$
\item \textbf{Non-negative orthant bounds}: If $\bx, \by \in \IRplus^p$ are non-negative vectors, then
\[
0 \le \mathsf{sim}(\bx, \by) \le 1
\]
Furthermore, for $\bx, \by \in \IRplus^p$,
\[
\mathsf{sim}(\bx, \by) = 0 \iff \mathsf{supp}(\bx) \cap \mathsf{supp}(\by) = \emptyset
\]
\end{enumerate}
\end{proposition}
\end{frame}


\begin{frame}[red]{Proof of Proposition~\ref{prop:cosine_properties} (Part 1: scale invariance)}
Let $\bx, \by \in \IR^p \setminus \{\b0\}$ and let $\alpha, \beta > 0$. Using the bilinearity of the dot product and the positive homogeneity of the Euclidean norm ($\|\alpha \bx\|_2 = |\alpha|\,\|\bx\|_2 = \alpha \|\bx\|_2$ since $\alpha > 0$), we compute
\begin{align*}
\mathsf{sim}(\alpha \bx, \beta \by)
&= \frac{(\alpha \bx) \bullet (\beta \by)}{\|\alpha \bx\|_2 \, \|\beta \by\|_2} \\
&= \frac{\alpha \beta (\bx \bullet \by)}{(\alpha \|\bx\|_2)(\beta \|\by\|_2)} \\
&= \frac{\cancel{\alpha \beta}(\bx \bullet \by)}{\cancel{\alpha \beta}\|\bx\|_2 \, \|\by\|_2} \\
&= \mathsf{sim}(\bx, \by)
\end{align*}
\vfill
(Note: if $\alpha < 0$ and $\beta > 0$, then $|\alpha| = -\alpha$, so $\mathsf{sim}(\alpha\bx, \beta\by) = -\mathsf{sim}(\bx, \by)$)\hfill\qed
\end{frame}


\begin{frame}[red]{Proof of Proposition~\ref{prop:cosine_properties} (Part 2: non-negative vectors)}
Assume $\bx, \by \in \IRplus^p \setminus \{\b0\}$, so $x_j \ge 0$ and $y_j \ge 0$ for every $j \in \{1, \ldots, p\}$
\vfill
\begin{enumerate}
\item Since the product of two non-negative real numbers is non-negative, $x_j y_j \ge 0$ for all $j = 1, \ldots, p$. Summing over $j$ yields
\[
\bx \bullet \by = \sum_{j=1}^p x_j y_j \ge 0
\]
Dividing by $\|\bx\|_2 \|\by\|_2 > 0$ and combining with Cauchy-Schwarz gives $0 \le \mathsf{sim}(\bx, \by) \le 1$
\vfill
\item Because each summand $x_j y_j$ is non-negative, a sum of non-negative terms $\sum_{j=1}^p x_j y_j$ equals $0$ if and only if \textbf{every individual term is zero}:
\[
\mathsf{sim}(\bx, \by) = 0 \iff \sum_{j=1}^p x_j y_j = 0 \iff \forall j \in \{1,\ldots,p\},\; x_j y_j = 0
\]
Since $x_j y_j = 0 \iff (x_j = 0 \textrm{ or } y_j = 0)$, there is no index $j$ with both $x_j > 0$ and $y_j > 0$, which means $\mathsf{supp}(\bx) \cap \mathsf{supp}(\by) = \emptyset$ \hfill\qed
\end{enumerate}
\end{frame}


\begin{frame}{Interpretation of Proposition~\ref{prop:cosine_properties} for text}
\begin{itemize}
\item Because bag-of-words and TF-IDF vectors lie in the non-negative orthant $\IRplus^p$, two documents can \textbf{never} have negative cosine similarity ($\theta \in [0, \pi/2]$)
\vfill
\item $\mathsf{sim}(\bx_i, \bx_k) = 0$ ($\theta = \pi/2$, orthogonal vectors) occurs \textbf{if and only if the two documents share no vocabulary words in common} (after any zero-IDF stop words are removed)
\vfill
\item Moreover, for $\ell_2$-normalised vectors $\hat{\bx} = \bx/\|\bx\|_2$ and $\hat{\by} = \by/\|\by\|_2$ on the unit sphere $S^{p-1}$, Euclidean distance and cosine similarity are monotonically related:
\[
\|\hat{\bx} - \hat{\by}\|_2^2 = \|\hat{\bx}\|_2^2 + \|\hat{\by}\|_2^2 - 2(\hat{\bx} \bullet \hat{\by}) = 2\left(1 - \mathsf{sim}(\bx, \by)\right)
\]
\end{itemize}
\end{frame}


\begin{frame}{All-pairs cosine similarity via the Gram matrix}
How do we compute the cosine similarity between \textbf{all pairs} of $n$ documents in a data matrix $X \in \M_{np}$ with rows $\bx_1^T, \ldots, \bx_n^T$ at once?
\vfill
\begin{importanttheorem}[Cosine similarity Gram matrix]\label{th:cosine_gram}
Let $X \in \M_{np}$ have non-zero document rows $\bx_1^T, \ldots, \bx_n^T \in \IR^{1 \times p}$ and let
\[
\hat{X} = D_{\text{norm}}^{-1} X = \begin{pmatrix} \hat{\bx}_1^T \\ \hat{\bx}_2^T \\ \vdots \\ \hat{\bx}_n^T \end{pmatrix} \in \M_{np}
\quad\textrm{where}\quad
D_{\text{norm}}^{-1} = \diag\!\left(\frac{1}{\|\bx_1\|_2}, \ldots, \frac{1}{\|\bx_n\|_2}\right)
\]
Then the symmetric \defword{Gram matrix}
\[
S = \hat{X} \hat{X}^T \in \M_n(\IR)
\]
has $(i,j)$ entry equal to the pairwise cosine similarity $s_{ij} = \hat{\bx}_i^T \hat{\bx}_j = \mathsf{sim}(\bx_i, \bx_j)$, with $s_{ii} = 1$ along the diagonal
\end{importanttheorem}
\end{frame}


\begin{frame}{Example -- Pairwise cosine similarity by hand (Part 1)}
Consider $n=3$ sparse document vectors in $\IRplus^5$ ($p = 5$):
\[
\bx_1 = \begin{pmatrix} 3 \\ 4 \\ 0 \\ 0 \\ 0 \end{pmatrix}, \qquad
\bx_2 = \begin{pmatrix} 6 \\ 8 \\ 0 \\ 0 \\ 0 \end{pmatrix}, \qquad
\bx_3 = \begin{pmatrix} 0 \\ 2 \\ 0 \\ 2 \\ 1 \end{pmatrix}
\]
\vfill
\textbf{Step 1: Compute the Euclidean norm of each document vector}:
\begin{align*}
\|\bx_1\|_2 &= \sqrt{3^2 + 4^2 + 0 + 0 + 0} = \sqrt{25} = 5 \\
\|\bx_2\|_2 &= \sqrt{6^2 + 8^2 + 0 + 0 + 0} = \sqrt{100} = 10 \\
\|\bx_3\|_2 &= \sqrt{0 + 2^2 + 0 + 2^2 + 1^2} = \sqrt{9} = 3
\end{align*}
\end{frame}


\begin{frame}{Example -- Pairwise cosine similarity by hand (Part 2)}
\textbf{Step 2: Compute pairwise dot products (only shared non-zero indices contribute!)}:
\begin{align*}
\bx_1 \bullet \bx_2 &= 3(6) + 4(8) = 18 + 32 = 50 \\
\bx_1 \bullet \bx_3 &= 4(2) = 8 \\
\bx_2 \bullet \bx_3 &= 8(2) = 16
\end{align*}
\vfill
\textbf{Step 3: Divide by norms to form the Gram matrix $S = \hat{X} \hat{X}^T$}:
\[
\mathsf{sim}(\bx_1, \bx_2) = \frac{50}{5 \times 10} = 1, \quad
\mathsf{sim}(\bx_1, \bx_3) = \frac{8}{5 \times 3} = \frac{8}{15}, \quad
\mathsf{sim}(\bx_2, \bx_3) = \frac{16}{10 \times 3} = \frac{8}{15}
\]
\[
\implies S = \hat{X} \hat{X}^T =
\begin{pmatrix}
1 & 1 & 8/15 \\
1 & 1 & 8/15 \\
8/15 & 8/15 & 1
\end{pmatrix}
\approx
\begin{pmatrix}
1.000 & 1.000 & 0.533 \\
1.000 & 1.000 & 0.533 \\
0.533 & 0.533 & 1.000
\end{pmatrix}
\]
\end{frame}

\SSsubsection{SVD vs NMF on non-negative data}

\begin{frame}{Recall from Unit 04: low-rank approximation via SVD}
In Unit 04, we saw that any matrix $X \in \M_{np}(\IR)$ admits a Singular Value Decomposition $X = U \Sigma V^T = \sum_{r=1}^{\rank(X)} \sigma_r \bu_r \bv_r^T$
\vfill
\begin{itemize}
\item By the \textbf{Eckart-Young theorem}, the best rank-$k$ approximation to $X$ in the Frobenius norm $\|A\|_F = \sqrt{\sum_{i=1}^n \sum_{j=1}^p a_{ij}^2}$ is the \defword{truncated SVD}:
\[
X_k = \sum_{r=1}^k \sigma_r \bu_r \bv_r^T = U_k \Sigma_k V_k^T
\]
\vfill
\item Applied to a document-term matrix $X$, truncated SVD is known in information retrieval as \defword{Latent Semantic Analysis} (\defword{LSA}, Deerwester et al.\ 1990)
\end{itemize}
\end{frame}


\begin{frame}{Why truncated SVD is awkward on non-negative data}
Many real-world data matrices are inherently \textbf{non-negative} ($x_{ij} \ge 0$):
\begin{itemize}
\item Document-term word count and TF-IDF matrices
\item Pixel intensities in greyscale or colour images
\item Audio spectrograms, chemical concentrations and gene expression counts
\end{itemize}
\vfill
Why does truncated SVD struggle with interpretability on $X \in \M_{np}(\IRplus)$?
\begin{itemize}
\item By the Perron-Frobenius theorem, the first singular vectors $\bu_1, \bv_1$ can be chosen non-negative
\item However, $U_k$ and $V_k$ have \textbf{orthonormal columns}: for every $r \ge 2$,
\[
\bu_r \bullet \bu_1 = 0 \quad\textrm{and}\quad \bv_r \bullet \bv_1 = 0
\]
\item As proved in Proposition~\ref{prop:cosine_properties}, a non-zero vector orthogonal to a strictly positive vector $\bv_1$ \textbf{must contain both positive and negative entries}!
\end{itemize}
\end{frame}


\begin{frame}{Subtractive cancellation vs additive parts}
\begin{itemize}
\item Because $\bu_2, \ldots, \bu_k$ and $\bv_2, \ldots, \bv_k$ contain negative numbers, SVD reconstructs a document $\bx_i \approx \sum_{r=1}^k (\sigma_r u_{ir}) \bv_r$ through \textbf{subtractive cancellation}:
\begin{itemize}
\item What does it mean for a ``topic'' vector $\bv_2$ to contain $-0.54 \times \code{"matrix"}$?
\item What does it mean to subtract $0.3$ of Topic 2 from a document so that negative word counts cancel out?
\end{itemize}
\vfill
\item For example, on our $4 \times 6$ document-term matrix $X$, the first two right singular vectors (term-topic directions) in $V_2$ are
\[
\bv_1 = \begin{pmatrix} 0.30 \\ 0.30 \\ 0.11 \\ 0.32 \\ 0.31 \\ 0.78 \end{pmatrix} \ge \b0, \qquad
\bv_2 = \begin{pmatrix} \boldred{-0.54} \\ \boldred{-0.47} \\ \boldred{-0.24} \\ 0.45 \\ 0.45 \\ 0.14 \end{pmatrix} \notin \IRplus^6
\]
\item To obtain directly interpretable topics (or parts of an image), we drop the orthogonality constraint and instead enforce \defword{non-negativity} on both factors!
\end{itemize}
\end{frame}


\begin{frame}{Non-negative Matrix Factorisation (NMF)}
\begin{importantdefinition}[Non-negative Matrix Factorisation (NMF)]
Given a non-negative data matrix $X \in \M_{np}(\IRplus)$ and a target rank $k < \min(n, p)$, \defword{Non-negative Matrix Factorisation} (\defword{NMF}, Paatero \& Tapper 1994; Lee \& Seung 1999) seeks two non-negative matrices
\[
W \in \M_{nk}(\IRplus) \quad\textrm{and}\quad H \in \M_{kp}(\IRplus)
\]
such that $X \approx W H$, by solving the constrained optimisation problem
\[
\min_{W \in \M_{nk}(\IRplus),\; H \in \M_{kp}(\IRplus)} \|X - W H\|_F^2
= \sum_{i=1}^n \sum_{j=1}^p \left(x_{ij} - \sum_{r=1}^k w_{ir} h_{rj}\right)^2
\]
subject to $w_{ir} \ge 0$ and $h_{rj} \ge 0$ for all $i, r, j$
\end{importantdefinition}
\end{frame}

\begin{itemize}
\item When $k \ll \min(n, p)$, NMF compresses $np$ entries into $k(n + p)$ non-negative parameters while preserving an \textbf{additive, parts-based} structure
\end{itemize}
\end{frame}

\SSsubsection{Topic modelling interpretation of W and H}

\begin{frame}{Row-wise view: documents as additive mixtures of topics}
Write $W = \begin{bmatrix} \bw_1^T \\ \vdots \\ \bw_n^T \end{bmatrix} \in \M_{nk}(\IRplus)$ in terms of its rows $\bw_i^T \in \IRplus^{1 \times k}$ and $H = \begin{bmatrix} \bh_1^T \\ \vdots \\ \bh_k^T \end{bmatrix} \in \M_{kp}(\IRplus)$ in terms of its rows $\bh_r^T \in \IRplus^{1 \times p}$
\vfill
By standard row-wise matrix multiplication, the $i$-th row of $X \approx W H$ is $\bx_i^T \approx \bw_i^T H = \sum_{r=1}^k w_{ir} \bh_r^T$, or in column-vector form:
\[
\bx_i \approx H^T \bw_i = w_{i1} \bh_1 + w_{i2} \bh_2 + \cdots + w_{ik} \bh_k = \sum_{r=1}^k w_{ir} \bh_r
\]
\vfill
\begin{itemize}
\item \textbf{Rows of $H$ ($\bh_r^T \in \IRplus^{1 \times p}$, the dictionary / topic basis)}:
\begin{itemize}
\item Row $\bh_r^T$ gives the weight of each of the $p$ vocabulary words in \defword{topic $r$}
\item The largest entries of $\bh_r$ identify the signature keywords of topic $r$
\end{itemize}
\vfill
\item \textbf{Rows of $W$ ($\bw_i^T \in \IRplus^{1 \times k}$, the document topic encodings)}:
\begin{itemize}
\item Entry $w_{ir} \ge 0$ tells how strongly document $d_i$ expresses topic $r$
\end{itemize}
\end{itemize}
\end{frame}

\begin{frame}{Rank-one outer product expansion and polysemy}
Just as the SVD writes $X_k = \sum_{r=1}^k \sigma_r \bu_r \bv_r^T$ as a sum of $k$ rank-one matrices, NMF decomposes $W H$ into $k$ \textbf{non-negative rank-one layers}:
\[
W H = \sum_{r=1}^k \bu_r \bh_r^T
\]
where $\bu_r \in \IRplus^n$ is the $r$-th column of $W$ and $\bh_r^T \in \IRplus^{1 \times p}$ is the $r$-th row of $H$
\vfill
\begin{itemize}
\item Each outer product $M_r = \bu_r \bh_r^T \in \M_{np}(\IRplus)$ displays the contribution of \textbf{topic $r$ alone} across all $n$ documents and all $p$ words
\item Because rows of $H$ do not have to be orthogonal, a single word $w_j$ with multiple meanings (\defword{polysemy}, e.g., \code{"degree"} in polynomial algebra vs vertex degree in graphs) can have positive weight $h_{rj} > 0$ in \textbf{more than one topic row}!
\end{itemize}
\end{frame}

\SSsubsection{Alternating least squares (ALS)}

\begin{frame}{Why computing NMF is an optimisation challenge}
Consider the NMF objective function
\[
f(W, H) = \|X - W H\|_F^2 = \sum_{i=1}^n \sum_{j=1}^p \left(x_{ij} - \sum_{r=1}^k w_{ir} h_{rj}\right)^2
\]
\vfill
\begin{enumerate}
\item \textbf{Non-uniqueness (scaling and permutation ambiguity)}:
\begin{itemize}
\item For any permutation matrix $P \in \M_k$ and any positive diagonal matrix $D = \diag(d_1, \ldots, d_k)$ with $d_r > 0$, setting $\tilde{W} = W P D \ge 0$ and $\tilde{H} = D^{-1} P^T H \ge 0$ gives $\tilde{W}\tilde{H} = W H$
\end{itemize}
\vfill
\item \textbf{Non-convexity jointly in $(W, H)$}:
\begin{itemize}
\item Because unknown parameters $w_{ir}$ and $h_{rj}$ are multiplied together in $W H$, $f(W, H)$ is \textbf{not convex} in $(W, H)$ simultaneously and can have multiple local minima
\end{itemize}
\end{enumerate}
\end{frame}


\begin{frame}{The key linear algebra insight: bi-convexity}
Although $f(W, H) = \|X - W H\|_F^2$ is non-convex jointly in $(W, H)$, it is \defword{bi-convex} (convex in $H$ when $W$ is fixed and convex in $W$ when $H$ is fixed)!
\vfill
\begin{enumerate}
\item \textbf{Hold $W \in \M_{nk}(\IRplus)$ fixed}: Denoting the $j$-th columns of $X$ and $H$ by $\bc_j \in \IRplus^n$ and $\bh_{\cdot j} \in \IRplus^k$,
\[
\|X - W H\|_F^2 = \sum_{j=1}^p \|\bc_j - W \bh_{\cdot j}\|_2^2
\]
Each term $\|\bc_j - W \bh_{\cdot j}\|_2^2$ is a standard \textbf{linear least-squares problem} $W \bh_{\cdot j} \approx \bc_j$ of the exact form studied in Unit 03 and Unit 04!
\vfill
\item \textbf{Hold $H \in \M_{kp}(\IRplus)$ fixed}: Using $\|A\|_F^2 = \|A^T\|_F^2$ and $(WH)^T = H^T W^T$,
\[
\|X - W H\|_F^2 = \|X^T - H^T W^T\|_F^2 = \sum_{i=1}^n \|\bx_i - H^T \bw_i\|_2^2
\]
This is again a linear least-squares problem for the rows $\bw_i^T$ of $W$ (columns $\bw_i$ of $W^T$) with design matrix $H^T \in \M_{pk}$!
\end{enumerate}
\end{frame}


\begin{frame}{Deriving the projected least-squares update for $H$}
\begin{proposition}[Projected least-squares step for $H$]\label{prop:als_H_step}
Suppose $W \in \M_{nk}(\IRplus)$ is fixed and has full column rank ($\rank(W) = k$, so $W^T W \in \M_k$ is invertible). Then:
\begin{enumerate}
\item For each column $j \in \{1, \ldots, p\}$, the unconstrained least-squares problem $\min_{\bh_{\cdot j} \in \IR^k} \|\bc_j - W \bh_{\cdot j}\|_2^2$ has the unique normal-equations solution
\[
\bh_{\cdot j}^{\text{LS}} = (W^T W)^{-1} W^T \bc_j
\]
\item Stacking all $p$ columns side by side into $H_{\text{LS}} = [\bh_{\cdot 1}^{\text{LS}} \cdots \bh_{\cdot p}^{\text{LS}}] \in \M_{kp}(\IR)$ gives the matrix formula
\[
H_{\text{LS}} = (W^T W)^{-1} W^T X
\]
\item In projected \defword{Alternating Least Squares} (\defword{ALS}), we enforce non-negativity by clipping negative entries to zero entrywise:
\[
H \leftarrow \max(0, H_{\text{LS}}) \quad\textrm{i.e.,}\quad h_{rj} = \max\!\left(0, (H_{\text{LS}})_{rj}\right)
\]
\end{enumerate}
\end{proposition}
\end{frame}


\begin{frame}{Deriving the projected least-squares update for $W$}
\begin{proposition}[Projected least-squares step for $W$]\label{prop:als_W_step}
Suppose $H \in \M_{kp}(\IRplus)$ is fixed and has full row rank ($\rank(H) = k$, so $H H^T \in \M_k$ is invertible). Then:
\begin{enumerate}
\item Minimising $\|X - W H\|_F^2 = \|X^T - H^T W^T\|_F^2$ without sign constraints yields the normal equations for $W^T$:
\[
(H^T)^T (H^T) W_{\text{LS}}^T = (H^T)^T X^T \iff (H H^T) W_{\text{LS}}^T = H X^T
\]
\item Solving for $W_{\text{LS}}^T$ and taking the transpose (using $((H H^T)^{-1})^T = (H H^T)^{-1}$) gives
\[
W_{\text{LS}}^T = (H H^T)^{-1} H X^T \iff W_{\text{LS}} = X H^T (H H^T)^{-1}
\]
\item Clipping negative entries to zero entrywise gives the projected ALS update:
\[
W \leftarrow \max(0, W_{\text{LS}})
\]
\end{enumerate}
\end{proposition}
\end{frame}


\begin{frame}{The Alternating Least Squares (ALS) algorithm for NMF}
Putting Propositions~\ref{prop:als_H_step} and~\ref{prop:als_W_step} together yields the \defword{Projected Alternating Least Squares} (\defword{ALS}) algorithm (Paatero \& Tapper 1994; Berry et al.\ 2007):
\vfill
\begin{enumerate}
\item \textbf{Input}: Non-negative data matrix $X \in \M_{np}(\IRplus)$ and target rank $k < \min(n, p)$
\item \textbf{Initialise}: Choose a non-negative initial matrix $W^{(0)} \in \M_{nk}(\IRplus)$ with $\rank(W^{(0)}) = k$ (e.g., using positive random numbers)
\item \textbf{Repeat for $t = 0, 1, 2, \ldots$ until convergence}:
\begin{enumerate}
\item \textbf{Update $H$ (fixing $W^{(t)}$)}: Solve $(W^{(t)T} W^{(t)}) H_{\text{LS}} = W^{(t)T} X$ and project:
\[
H^{(t+1)} = \max\!\left(0,\; (W^{(t)T} W^{(t)})^{-1} W^{(t)T} X\right)
\]
\item \textbf{Update $W$ (fixing $H^{(t+1)}$)}: Solve $W_{\text{LS}} (H^{(t+1)} H^{(t+1)T}) = X H^{(t+1)T}$ and project:
\[
W^{(t+1)} = \max\!\left(0,\; X H^{(t+1)T} (H^{(t+1)} H^{(t+1)T})^{-1}\right)
\]
\item \emph{(Optional)} Normalise each row of $H^{(t+1)}$ (or column of $W^{(t+1)}$) to unit $\ell_2$ or $\ell_1$ norm to prevent scaling drift
\end{enumerate}
\end{enumerate}
\end{frame}


\begin{frame}{Example -- One ALS update step by hand (Part 1)}
Let $X \in \M_{33}(\IRplus)$ and let the initial rank-$2$ factor matrix $W^{(0)} \in \M_{32}(\IRplus)$ be
\[
X = \begin{pmatrix} 3 & 0 & 2 \\ 0 & 3 & 3 \\ 0 & 3 & 1 \end{pmatrix}, \qquad
W^{(0)} = \begin{pmatrix} 1 & 0 \\ 1 & 1 \\ 0 & 1 \end{pmatrix}
\]
Let us compute the first ALS update $H^{(1)} = \max\!\left(0, (W^{(0)T} W^{(0)})^{-1} W^{(0)T} X\right)$ by hand
\vfill
\textbf{Step 1: Compute $W^{(0)T} W^{(0)} \in \M_2$ and its inverse}:
\[
W^{(0)T} W^{(0)} =
\begin{pmatrix} 1 & 1 & 0 \\ 0 & 1 & 1 \end{pmatrix}
\begin{pmatrix} 1 & 0 \\ 1 & 1 \\ 0 & 1 \end{pmatrix}
= \begin{pmatrix} 2 & 1 \\ 1 & 2 \end{pmatrix}
\]
Since $\det(W^{(0)T} W^{(0)}) = 2(2) - 1(1) = 3 \neq 0$, the $2 \times 2$ inverse formula gives
\[
(W^{(0)T} W^{(0)})^{-1} = \frac{1}{3} \begin{pmatrix} 2 & -1 \\ -1 & 2 \end{pmatrix}
\]
\end{frame}


\begin{frame}{Example -- One ALS update step by hand (Part 2)}
\textbf{Step 2: Compute $W^{(0)T} X \in \M_{23}$}:
\[
W^{(0)T} X =
\begin{pmatrix} 1 & 1 & 0 \\ 0 & 1 & 1 \end{pmatrix}
\begin{pmatrix} 3 & 0 & 2 \\ 0 & 3 & 3 \\ 0 & 3 & 1 \end{pmatrix}
= \begin{pmatrix} 3 & 3 & 5 \\ 0 & 6 & 4 \end{pmatrix}
\]
\vfill
\textbf{Step 3: Compute the unconstrained least-squares matrix $H_{\text{LS}}^{(1)}$}:
\[
H_{\text{LS}}^{(1)} = (W^{(0)T} W^{(0)})^{-1} W^{(0)T} X
= \frac{1}{3} \begin{pmatrix} 2 & -1 \\ -1 & 2 \end{pmatrix} \begin{pmatrix} 3 & 3 & 5 \\ 0 & 6 & 4 \end{pmatrix}
= \frac{1}{3} \begin{pmatrix} 6 & 0 & 6 \\ -3 & 9 & 3 \end{pmatrix}
= \begin{pmatrix} 2 & 0 & 2 \\ \boldred{-1} & 3 & 1 \end{pmatrix}
\]
\vfill
\textbf{Step 4: Project onto $\IRplus$ by clipping negative entries to zero}:
\[
H^{(1)} = \max\!\left(0, H_{\text{LS}}^{(1)}\right) = \begin{pmatrix} 2 & 0 & 2 \\ \boldred{0} & 3 & 1 \end{pmatrix} \in \M_{23}(\IRplus)
\]
Notice how unconstrained least squares produced $-1$ in entry $(2,1)$, which the projection step clips to $0$!
\end{frame}


\begin{frame}{Example -- Evaluating the reconstruction $W^{(0)} H^{(1)}$ (Part 3)}
\textbf{Step 5: Compute the rank-$2$ approximation $\hat{X}^{(1)} = W^{(0)} H^{(1)}$ and its Frobenius error}:
\[
\hat{X}^{(1)} = W^{(0)} H^{(1)} =
\begin{pmatrix} 1 & 0 \\ 1 & 1 \\ 0 & 1 \end{pmatrix}
\begin{pmatrix} 2 & 0 & 2 \\ 0 & 3 & 1 \end{pmatrix}
= \begin{pmatrix} 2 & 0 & 2 \\ 2 & 3 & 3 \\ 0 & 3 & 1 \end{pmatrix}
\]
\vfill
Comparing $\hat{X}^{(1)}$ with $X = \begin{pmatrix} 3 & 0 & 2 \\ 0 & 3 & 3 \\ 0 & 3 & 1 \end{pmatrix}$, the residual matrix is
\[
X - W^{(0)} H^{(1)} = \begin{pmatrix} 1 & 0 & 0 \\ -2 & 0 & 0 \\ 0 & 0 & 0 \end{pmatrix}
\implies
\|X - W^{(0)} H^{(1)}\|_F^2 = 1^2 + (-2)^2 = 5
\]
\vfill
\begin{itemize}
\item Columns 2 and 3 of $X$ are reconstructed \textbf{with zero error} on the very first half-step; the next half-step updates $W^{(1)} = \max\!\left(0, X H^{(1)T}(H^{(1)}H^{(1)T})^{-1}\right)$ to reduce the error on column 1 even further!
\end{itemize}
\end{frame}


\begin{frame}{Remark: exact NNLS and multiplicative updates}
\begin{itemize}
\item \textbf{Projected ALS vs exact Non-Negative Least Squares (NNLS)}:
\begin{itemize}
\item Solving unconstrained least squares and clipping $\max(0, \cdot)$ is fast (requiring only a $k \times k$ linear system $W^T W$ per iteration) and works well when $k$ is small and $X$ is sparse
\item A more accurate variant solves each subproblem $\min_{\bw_i \ge \b0} \|\bx_i - H^T \bw_i\|_2^2$ and $\min_{\bh_{\cdot j} \ge \b0} \|\bc_j - W \bh_{\cdot j}\|_2^2$ using an exact active-set \defword{NNLS} solver (Lawson \& Hanson 1974), guaranteeing that $\|X - W^{(t)}H^{(t)}\|_F^2$ decreases monotonically at every step
\end{itemize}
\vfill
\item \textbf{Lee \& Seung (1999) multiplicative updates}:
\begin{itemize}
\item Another popular algorithm updates entries by elementwise multiplication:
\[
h_{rj} \leftarrow h_{rj} \frac{(W^T X)_{rj}}{(W^T W H)_{rj}}, \qquad
w_{ir} \leftarrow w_{ir} \frac{(X H^T)_{ir}}{(W H H^T)_{ir}}
\]
\item Because all factors on the right-hand side are non-negative, $W$ and $H$ automatically remain non-negative without clipping while $\|X - WH\|_F^2$ is guaranteed not to increase
\end{itemize}
\end{itemize}
\end{frame}

\begin{frame}{Summary of Unit U11}
\begin{itemize}
\item \textbf{Tokenisation and bag-of-words}: A vocabulary $V = \{w_1, \ldots, w_p\}$ maps each token to $\be_j \in \IR^p$ and each document $d_i$ to a sparse count vector $\bx_i = \sum_{j=1}^p x_{ij} \be_j \in \IRplus^p$ with $\|\bx_i\|_0 \ll p$
\vfill
\item \textbf{Document-term matrices and TF-IDF scaling}: Stacking document vectors as rows forms $X \in \M_{np}(\IRplus)$ where TF-IDF with length normalisation is the two-sided diagonal matrix product
\[
\hat{X} = D_{\text{len}}^{-1} X D_{\text{IDF}}
\]
\vfill
\item \textbf{Cosine similarity in high dimensions}: Semantic similarity is measured by the scale-invariant angle $\mathsf{sim}(\bx, \by) = \frac{\bx \bullet \by}{\|\bx\|_2 \|\by\|_2} \in [0, 1]$ on $\IRplus^p$ whose pairwise values form the Gram matrix $S = \hat{X} \hat{X}^T \in \M_n(\IR)$
\vfill
\item \textbf{Non-negative Matrix Factorisation (NMF)}: Approximating $X \approx W H$ with $W \in \M_{nk}(\IRplus)$ and $H \in \M_{kp}(\IRplus)$ avoids the subtractive cancellation of SVD, yielding additive topic rows $\bh_r^T$ and document encodings $\bw_i^T$ computed via projected \textbf{Alternating Least Squares (ALS)}
\end{itemize}
\end{frame}
